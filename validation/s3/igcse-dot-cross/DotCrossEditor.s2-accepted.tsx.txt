import {useEffect,useMemo,useRef,useState} from 'react';
import type {KeyboardEvent,PointerEvent} from 'react';
import type {DotCrossState,DotCrossElement,DotCrossAnchor,ElectronSymbol,Point} from '../../contracts/index.ts';
import type {EditorSurfaceProps} from '../../ui/EditorFrame.tsx';
import {dotCrossEngine,type DotCrossCommand} from '../../chemistry/dot-and-cross/engine.ts';
import {createLayout} from '../../chemistry/dot-and-cross/layout.js';
import './editor.css';
const elements:readonly DotCrossElement[]=['H','B','C','Si','N','O','P','S','F','Cl','Br','I','Li','Na','K','Mg','Ca','Al'];
const symbols:readonly ElectronSymbol[]=['dot','cross','triangle'];
type Tool='select'|'atom'|'dot'|'cross'|'triangle'|'erase';
const pointIn=(event:PointerEvent<SVGSVGElement>,svg:SVGSVGElement):Point=>{const matrix=svg.getScreenCTM();if(!matrix)return {x:500,y:325};const p=new DOMPoint(event.clientX,event.clientY).matrixTransform(matrix.inverse());return {x:p.x,y:p.y};};
/** Meaningful semantic edits only are returned to the shared attempt controller. */
export function DotCrossEditor({part,response,readOnly,onResponse}:EditorSurfaceProps){
 if(part.kind!=='dot-and-cross')throw Error('DotCrossEditor needs a dot-and-cross part.');
 const state:DotCrossState=response?.kind==='dot-and-cross'?response:part.initial;
 const questionId=part.markingPolicyId.replace(/^dot-cross:/,'');
 const layout=useMemo(()=>createLayout(questionId),[questionId]);
 const [tool,setTool]=useState<Tool>('atom'),[element,setElement]=useState<DotCrossElement>('H');
 const [selected,setSelected]=useState<readonly string[]>([]),[symbol,setSymbol]=useState<ElectronSymbol>('dot');
 const [charge,setCharge]=useState(1),[bracket,setBracket]=useState(true),[message,setMessage]=useState('Choose an atom, then tap the canvas or use Add atom.');
 const [x,setX]=useState(400),[y,setY]=useState(325),[first,setFirst]=useState(''),[second,setSecond]=useState(''),[slot,setSlot]=useState(0);
 const [magnified,setMagnified]=useState(false),viewport=useRef<HTMLDivElement>(null);
 useEffect(()=>{const node=viewport.current;if(node)node.scrollLeft=magnified?Math.max(0,(node.scrollWidth-node.clientWidth)/2):0;},[magnified]);
 const drag=useRef<{start:Point;id:string}|null>(null),[preview,setPreview]=useState<{dx:number;dy:number}|null>(null);
 const liveSelected=selected.filter(id=>state.atoms.some(atom=>atom.id===id));
 const apply=(command:DotCrossCommand)=>{if(readOnly)return;const next=dotCrossEngine.apply(state,command);if(next===state){setMessage('No change: that electron position is occupied or the selection is incomplete.');return;}onResponse(next);setMessage('Diagram updated.');};
 const addElectron=(anchor:DotCrossAnchor)=>apply({type:'electron',symbol:tool==='cross'||tool==='triangle'?tool:tool==='dot'?'dot':symbol,anchor});
 const select=(id:string,multiple:boolean)=>{setTool('select');setSelected(current=>multiple?current.includes(id)?current.filter(value=>value!==id):[...current,id]:[id]);setFirst(id);};
 const shownState=preview&&drag.current?{...state,atoms:state.atoms.map(atom=>atom.id===drag.current?.id?{...atom,x:atom.x+preview.dx,y:atom.y+preview.dy}:atom)}:state;
 const onCanvasDown=(event:PointerEvent<SVGSVGElement>)=>{
  if(readOnly)return;const target=event.target as SVGElement;if(target.closest('[data-object]'))return;
  const p=pointIn(event,event.currentTarget);
  if(tool==='atom')apply({type:'atom',element,point:p});
  else if(symbols.includes(tool as ElectronSymbol)){const anchor=layout.regionAt(state,p);if(anchor)addElectron(anchor);else setMessage('Tap a shell or shared overlap, or choose atoms and an electron slot below.');}
  else setSelected([]);
 };
 const onAtomDown=(event:PointerEvent<SVGGElement>,id:string)=>{
  if(readOnly)return;event.stopPropagation();const atom=state.atoms.find(item=>item.id===id)!;
  if(tool==='erase'){apply({type:'delete',ids:[id]});return;}
  if(symbols.includes(tool as ElectronSymbol)){const svg=event.currentTarget.ownerSVGElement!;const p=pointIn(event as unknown as PointerEvent<SVGSVGElement>,svg);const anchor=layout.regionAt(state,p);if(anchor)addElectron(anchor);return;}
  select(id,event.shiftKey);if(tool==='select'){const svg=event.currentTarget.ownerSVGElement!;drag.current={start:pointIn(event as unknown as PointerEvent<SVGSVGElement>,svg),id};event.currentTarget.setPointerCapture(event.pointerId);setX(Math.round(atom.x));setY(Math.round(atom.y));}
 };
 const onAtomMove=(event:PointerEvent<SVGGElement>)=>{if(!drag.current)return;const svg=event.currentTarget.ownerSVGElement!,p=pointIn(event as unknown as PointerEvent<SVGSVGElement>,svg);setPreview({dx:p.x-drag.current.start.x,dy:p.y-drag.current.start.y});};
 const endDrag=()=>{if(drag.current&&preview)apply({type:'move',ids:[drag.current.id],dx:preview.dx,dy:preview.dy});drag.current=null;setPreview(null);};
 const keyboard=(event:KeyboardEvent<SVGSVGElement>)=>{if(readOnly)return;const key=event.key.toLowerCase();if((event.ctrlKey||event.metaKey)&&key==='z'){event.preventDefault();apply({type:event.shiftKey?'redo':'undo'});return;}
  if(['arrowleft','arrowright','arrowup','arrowdown'].includes(key)&&liveSelected.length){event.preventDefault();const step=event.shiftKey?20:5;apply({type:'move',ids:liveSelected,dx:key==='arrowleft'?-step:key==='arrowright'?step:0,dy:key==='arrowup'?-step:key==='arrowdown'?step:0});}
  if(key==='delete'&&liveSelected.length){event.preventDefault();apply({type:'delete',ids:liveSelected});setSelected([]);}
 };
 const manualElectron=()=>{if(!first){setMessage('Choose the first atom.');return;}if(second&&first===second){setMessage('Shared electrons need two different atoms.');return;}
  addElectron(second?{kind:'bond',a:first,b:second,slot:Math.min(5,slot) as 0|1|2|3|4|5}:{kind:'atom',atomId:first,slot:slot as 0|1|2|3|4|5|6|7});};
 const atomOptions=state.atoms.map(atom=><option key={atom.id} value={atom.id}>{atom.element} {atom.id}</option>);
 const controls=<div className="dc-controls"><fieldset disabled={readOnly}><legend>Drawing tools</legend><div className="dc-tools">{(['select','atom','dot','cross','triangle','erase'] as const).map(value=><button key={value} type="button" aria-pressed={tool===value} onClick={()=>setTool(value)}>{value.charAt(0).toUpperCase()+value.slice(1)}</button>)}<label>Element <select aria-label="Element" value={element} onChange={event=>setElement(event.target.value as DotCrossElement)}>{elements.map(value=><option key={value}>{value}</option>)}</select></label></div></fieldset>
  <div className="dc-tools"><button type="button" disabled={readOnly||!state.history?.length} onClick={()=>apply({type:'undo'})}>Undo</button><button type="button" disabled={readOnly||!state.future?.length} onClick={()=>apply({type:'redo'})}>Redo</button><button type="button" disabled={readOnly||!state.atoms.length} onClick={()=>apply({type:'clear'})}>Clear diagram</button></div></div>;
 const alternatives=<div className="dc-alternatives"><fieldset disabled={readOnly}><legend>Place atoms without dragging</legend><label>X position <input type="number" aria-label="X position" min="75" max="925" value={x} onChange={event=>setX(Number(event.target.value))}/></label><label>Y position <input type="number" aria-label="Y position" min="75" max="575" value={y} onChange={event=>setY(Number(event.target.value))}/></label><button type="button" onClick={()=>apply({type:'atom',element,point:{x,y}})}>Add atom</button>
  <div className="dc-atom-list" aria-label="Select atoms">{state.atoms.map(atom=><button type="button" key={atom.id} aria-pressed={liveSelected.includes(atom.id)} onClick={()=>{setSelected(current=>current.includes(atom.id)?current.filter(id=>id!==atom.id):[...current,atom.id]);setFirst(atom.id);}}>{atom.element} {atom.id}</button>)}</div>
  <div className="dc-tools">{([{label:'Move left',dx:-20,dy:0},{label:'Move right',dx:20,dy:0},{label:'Move up',dx:0,dy:-20},{label:'Move down',dx:0,dy:20}] as const).map(move=><button type="button" key={move.label} disabled={!liveSelected.length} onClick={()=>apply({type:'move',ids:liveSelected,dx:move.dx,dy:move.dy})}>{move.label}</button>)}<button type="button" disabled={!liveSelected.length} onClick={()=>apply({type:'delete',ids:liveSelected})}>Delete selected atoms</button></div></fieldset>
  <fieldset disabled={readOnly}><legend>Place an electron in a shell or bond</legend><label>Electron symbol <select aria-label="Electron symbol" value={symbol} onChange={event=>{setSymbol(event.target.value as ElectronSymbol);setTool('select');}}>{symbols.map(value=><option key={value}>{value}</option>)}</select></label><label>First atom <select aria-label="First atom" value={first} onChange={event=>setFirst(event.target.value)}><option value="">Choose atom</option>{atomOptions}</select></label><label>Second atom (shared bond) <select aria-label="Second atom" value={second} onChange={event=>setSecond(event.target.value)}><option value="">Lone electron</option>{atomOptions}</select></label><label>Electron slot <select aria-label="Electron slot" value={slot} onChange={event=>setSlot(Number(event.target.value))}>{Array.from({length:second?6:8},(_,index)=><option key={index} value={index}>{index+1}</option>)}</select></label><button type="button" onClick={manualElectron}>Add electron</button></fieldset>
  <fieldset disabled={readOnly}><legend>Bracket and charge selected atoms</legend><label>Ion charge <input type="number" aria-label="Ion charge" min="-9" max="9" step="1" value={charge} onChange={event=>setCharge(Number(event.target.value))}/></label><label><input type="checkbox" checked={bracket} onChange={event=>setBracket(event.target.checked)}/> Show brackets</label><button type="button" disabled={!liveSelected.length} onClick={()=>apply({type:'group',atomIds:liveSelected,charge,bracket})}>Apply bracket and charge</button></fieldset>
  <details><summary>Electron inventory and deletion</summary><ul>{state.electrons.map(electron=><li key={electron.id}>{electron.symbol} · {electron.anchor.kind==='atom'?electron.anchor.atomId:`${electron.anchor.a}–${electron.anchor.b}`} · slot {electron.anchor.slot+1} <button type="button" disabled={readOnly} aria-label={`Delete electron ${electron.id}`} onClick={()=>apply({type:'delete',ids:[electron.id]})}>Delete</button></li>)}</ul><ul>{state.groups.map(group=><li key={group.id}>{group.atomIds.join(', ')} {layout.chargeText(group.charge)} <button type="button" disabled={readOnly} aria-label={`Delete group ${group.id}`} onClick={()=>apply({type:'delete',ids:[group.id]})}>Remove bracket and charge</button></li>)}</ul></details>
  <p>Select an atom with Enter or the atom buttons. Arrow keys move the selection; Shift moves farther. Delete removes selected atoms; Ctrl+Z undoes. Touch users can tap slots or use these controls.</p></div>;
 return <div className="dot-cross-editor" data-question-id={questionId}>
  <p className="editor-instructions">Place atoms and electrons. Select atoms to move them or bracket an ion. Dots, crosses and triangles represent electron origins.</p>{controls}
  <div className="dc-view-controls"><button type="button" aria-pressed={!magnified} onClick={()=>setMagnified(false)}>Fit diagram</button><button type="button" aria-pressed={magnified} onClick={()=>setMagnified(true)}>Magnify diagram</button><span>Magnify for larger labels and electrons; scroll or swipe to inspect the drawing.</span></div>
  <div className="dc-viewport" ref={viewport} tabIndex={0} aria-label="Scrollable diagram viewport">
  <svg viewBox="0 0 1000 650" className="dc-canvas" style={{width:magnified?'1000px':'100%',minWidth:magnified?'1000px':'0',touchAction:magnified?'pan-x pan-y':'none'}} aria-label="Dot-and-cross drawing canvas" role="group" tabIndex={0} onKeyDown={keyboard} onPointerDown={onCanvasDown}>
   <title>Dot-and-cross diagram</title><rect x="0" y="0" width="1000" height="650" fill="white"/>
   {layout.bondPairs(shownState).map(([aId,bId])=>{const a=shownState.atoms.find(atom=>atom.id===aId)!,b=shownState.atoms.find(atom=>atom.id===bId)!;return <line key={`${aId}:${bId}`} x1={a.x} y1={a.y} x2={b.x} y2={b.y} stroke="#d1d5db" strokeWidth="1"/>;})}
   {shownState.atoms.map(atom=><g key={atom.id} data-object={atom.id} role="button" tabIndex={0} aria-label={`${atom.element} atom ${atom.id}`} aria-pressed={liveSelected.includes(atom.id)} onKeyDown={event=>{if(event.key==='Enter'||event.key===' '){event.preventDefault();select(atom.id,event.shiftKey);}}} onPointerDown={event=>onAtomDown(event,atom.id)} onPointerMove={onAtomMove} onPointerUp={endDrag} onPointerCancel={()=>{drag.current=null;setPreview(null);}}>
    <circle cx={atom.x} cy={atom.y} r={layout.shellRadius(atom)} fill={liveSelected.includes(atom.id)?'#eff6ff':'transparent'} stroke={liveSelected.includes(atom.id)?'#1d4ed8':'#94a3b8'} strokeWidth={liveSelected.includes(atom.id)?3:1.5}/><text x={atom.x} y={atom.y+8} textAnchor="middle" fontSize="25" fontWeight="bold">{atom.element}</text></g>)}
   {shownState.electrons.map(electron=>{const p=layout.point(shownState,electron.anchor);if(!p)return null;return <g key={electron.id} data-object={electron.id} role="button" tabIndex={readOnly?-1:0} aria-label={`${electron.symbol} electron ${electron.id}`} onPointerDown={event=>{event.stopPropagation();if(tool==='erase')apply({type:'delete',ids:[electron.id]});}} onKeyDown={event=>{if(event.key==='Delete')apply({type:'delete',ids:[electron.id]});}}><circle cx={p.x} cy={p.y} r="11" fill="transparent"/>{electron.symbol==='dot'?<circle cx={p.x} cy={p.y} r="3.4" fill="#111827"/>:electron.symbol==='cross'?<path d={`M${p.x-4} ${p.y-4}l8 8m-8 0l8 -8`} fill="none" stroke="#111827" strokeWidth="2"/>:<path d={`M${p.x} ${p.y-5}l5 9h-10Z`} fill="none" stroke="#111827" strokeWidth="2"/>}</g>;})}
   {shownState.groups.map(group=>{const b=layout.groupBounds(shownState,group.atomIds);if(!b)return null;return <g key={group.id} data-object={group.id} aria-label={`Ion ${group.atomIds.join(', ')} charge ${group.charge}`} onPointerDown={event=>{event.stopPropagation();if(tool==='erase')apply({type:'delete',ids:[group.id]});}}>{group.bracket&&<path d={`M${b.x+12} ${b.y}h-12v${b.bottom-b.y}h12M${b.right-12} ${b.y}h12v${b.bottom-b.y}h-12`} fill="none" stroke="#111827" strokeWidth="2.5"/>}<text x={b.right+7} y={b.y+8} fontSize="24">{layout.chargeText(group.charge)}</text></g>;})}
   {!readOnly&&symbols.includes(tool as ElectronSymbol)&&layout.targets(shownState,liveSelected).map(target=><circle key={layout.key(target.anchor)} cx={target.x} cy={target.y} r="8" fill="#dbeafe" stroke="#3b82f6" opacity="0.8" data-object="electron-slot" onPointerDown={event=>{event.stopPropagation();addElectron(target.anchor);}}/>)}
  </svg></div><p className="dc-status" role="status">{message}</p>
  <details className="editor-alternatives"><summary>Keyboard and non-drag controls</summary>{alternatives}</details>
 </div>;
}
