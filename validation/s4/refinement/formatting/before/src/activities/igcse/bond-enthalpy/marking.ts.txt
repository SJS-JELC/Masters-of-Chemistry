import type {MarkingPolicy} from '../../../contracts/index.ts';
import {bondQuestion,bondSource} from './provider.ts';
import {markNumbers,score} from '../../../chemistry/thermochemistry/marking.ts';
export const bondMarking:MarkingPolicy={id:'bond-source-numeric-and-self-drawing',mark(question,responses){const checked=bondQuestion(question.ref),source=bondSource(question.ref.questionId);if(JSON.stringify(question.parts.filter(p=>p.kind!=='drawing-self-check').map(p=>p.id))!==JSON.stringify(checked.parts.filter(p=>p.kind!=='drawing-self-check').map(p=>p.id)))throw Error('Bond response parts disagree with source');return markNumbers(source.responses,responses);},masteryScore:score};
