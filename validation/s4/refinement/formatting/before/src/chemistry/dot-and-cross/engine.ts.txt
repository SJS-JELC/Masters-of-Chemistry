import type {DotCrossState,DotCrossSnapshot,DotCrossElement,DotCrossAnchor,ElectronSymbol,EditorEngine,Point} from '../../contracts/index.ts';
import {validateState} from './core.js';
export type DotCrossCommand=
 | Readonly<{type:'atom';element:DotCrossElement;point:Point}>
 | Readonly<{type:'move';ids:readonly string[];dx:number;dy:number}>
 | Readonly<{type:'electron';symbol:ElectronSymbol;anchor:DotCrossAnchor}>
 | Readonly<{type:'delete';ids:readonly string[]}>
 | Readonly<{type:'group';atomIds:readonly string[];charge:number;bracket:boolean}>
 | Readonly<{type:'undo'|'redo'|'clear'}>;
export const emptyDiagram=():DotCrossState=>({kind:'dot-and-cross',atoms:[],electrons:[],groups:[]});
export const snapshot=(state:DotCrossSnapshot):DotCrossSnapshot=>({kind:'dot-and-cross',atoms:structuredClone(state.atoms),electrons:structuredClone(state.electrons),groups:structuredClone(state.groups)});
function nextId(state:DotCrossState,prefix:string){const ids=new Set([...state.atoms,...state.electrons,...state.groups].map(item=>item.id));let index=1;while(ids.has(`${prefix}${index}`))index++;return `${prefix}${index}`;}
function apply(state:DotCrossState,command:DotCrossCommand):DotCrossState{
 const history=state.history??[],future=state.future??[];
 if(command.type==='undo'){const previous=history.at(-1);return previous?{...snapshot(previous),history:history.slice(0,-1),future:[snapshot(state),...future].slice(0,100)}:state;}
 if(command.type==='redo'){const next=future[0];return next?{...snapshot(next),history:[...history,snapshot(state)].slice(-100),future:future.slice(1)}:state;}
 let changed:DotCrossSnapshot=snapshot(state);
 switch(command.type){
  case 'atom':changed={...changed,atoms:[...changed.atoms,{id:nextId(state,'a'),element:command.element,x:command.point.x,y:command.point.y}]};break;
  case 'move':changed={...changed,atoms:changed.atoms.map(atom=>command.ids.includes(atom.id)?{...atom,x:Math.max(75,Math.min(925,atom.x+command.dx)),y:Math.max(75,Math.min(575,atom.y+command.dy))}:atom)};break;
  case 'electron':changed={...changed,electrons:[...changed.electrons,{id:nextId(state,'e'),symbol:command.symbol,anchor:command.anchor}]};break;
  case 'delete':changed={...changed,atoms:changed.atoms.filter(atom=>!command.ids.includes(atom.id)),electrons:changed.electrons.filter(e=>!command.ids.includes(e.id)&&(e.anchor.kind==='atom'?!command.ids.includes(e.anchor.atomId):!command.ids.includes(e.anchor.a)&&!command.ids.includes(e.anchor.b))),groups:changed.groups.filter(g=>!command.ids.includes(g.id)&&!g.atomIds.some(id=>command.ids.includes(id)))};break;
  case 'group':changed={...changed,groups:[...changed.groups.filter(g=>!g.atomIds.some(id=>command.atomIds.includes(id))),{id:nextId(state,'g'),atomIds:[...command.atomIds],charge:command.charge,bracket:command.bracket}]};break;
  case 'clear':changed=emptyDiagram();break;
 }
 if(validateState(changed).length||JSON.stringify(snapshot(state))===JSON.stringify(changed))return state;
 return {...changed,history:[...history,snapshot(state)].slice(-100),future:[]};
}
export const dotCrossEngine:EditorEngine<'dot-and-cross',DotCrossCommand>={kind:'dot-and-cross',apply,
 checkSubmission(state){const errors=validateState(state);if(errors.length)return {status:'malformed',issues:errors.map(message=>({code:'reference-integrity',message,objectIds:[]}))};
  if(!state.atoms.length)return {status:'incomplete',issues:[{code:'empty',message:'Place atoms to begin the diagram.',objectIds:[]}]};
  // Excessive valence, wrong bonds, missing electrons and charges are assessable chemistry.
  return {status:'ready',chemicalIssues:[]};},
 validate(state){const errors=validateState(state);return errors.length?{valid:false,issues:errors.map(message=>({code:'reference-integrity',message,objectIds:[]}))}:{valid:true};}
};
