import type {Level} from '../../contracts/index.ts';
import {data} from './data.js';
import {coreOptions,same} from './core.js';
export const capacity=36**6;
export function hash(value:string){let n=2166136261;for(const c of value)n=Math.imul(n^c.charCodeAt(0),16777619)>>>0;return n;}
export function format(prefix:string,n:number){if(!Number.isSafeInteger(n)||n<0||n>=capacity)throw Error('Invalid review ID.');return `${prefix}-${n.toString(36).toUpperCase().padStart(6,'0')}`;}
export function parse(prefix:string,code:string){const match=new RegExp(`^${prefix}-([0-9A-Z]{6})$`).exec(code.trim().toUpperCase());if(!match)throw Error('Invalid review ID.');return parseInt(match[1]!,36);}
export interface MainVariant {kind:'main';speciesId:string;direction:'build'|'identify';representation:string;group:string;questionId:string;minimumLevel:Level}
export const variants:readonly MainVariant[]=data.species.flatMap(item=>data.representations.filter(rep=>rep.id!=='short'||coreOptions(item).length).flatMap(rep=>(['build','identify'] as const).map(direction=>({kind:'main' as const,speciesId:item.id,direction,representation:rep.id,group:item.group,questionId:format('EC',hash(`${item.id}:${direction}:${rep.id}`)%capacity),minimumLevel:(item.group.startsWith('d-')?3:item.group==='main-ions'||rep.id==='short'?2:1) as Level}))));
export interface MatchingVariant {kind:'bonus';questionId:string;counts:readonly number[];options:readonly string[];mask:number}
export const matchingFamilies=[...new Map(data.species.map(item=>[item.counts.join(','),data.species.filter(other=>same(item.counts,other.counts))])).values()].filter(bucket=>bucket.length>=2&&data.species.length-bucket.length>=3).map(bucket=>({templateId:`matching:${bucket[0]!.counts.join(',')}`,counts:bucket[0]!.counts}));
/** Exact ECB low four mask bits and source xorshift/shuffle; no legacy store/session. */
export function matchingFromId(code:string):MatchingVariant {
 let rng=parse('ECB',code);const mask=rng%16;
 const groups=data.groups.filter((_group,index)=>mask&2**index).map(group=>group.id);
 if(!groups.length)throw Error('This review ID has no species groups.');
 const items=data.species.filter(item=>groups.includes(item.group));
 function random(){let n=rng>>>0;n^=n<<13;n^=n>>>17;n^=n<<5;rng=n>>>0;return rng/4294967296;}
 function shuffle<T>(values:readonly T[]):T[]{const result=values.slice();for(let i=result.length-1;i>0;i--){const j=Math.floor(random()*(i+1));[result[i],result[j]]=[result[j]!,result[i]!];}return result;}
 const buckets=new Map<string,typeof items>();for(const item of items){const key=item.counts.join(',');if(!buckets.has(key))buckets.set(key,[]);buckets.get(key)!.push(item);}
 const eligible=[...buckets.values()].filter(bucket=>bucket.length>=2&&items.length-bucket.length>=3);
 if(!eligible.length)throw Error('No matching question is available for that ID.');
 const matches=shuffle(eligible)[0]!,positive=shuffle(matches).slice(0,3),negative=shuffle(items.filter(item=>!same(item.counts,matches[0]!.counts))).slice(0,3);
 return {kind:'bonus',questionId:format('ECB',parse('ECB',code)),counts:matches[0]!.counts.slice(),options:shuffle([...positive,...negative].map(item=>item.id)),mask};
}
export function getVariant(id:string):MainVariant|MatchingVariant{if(id.startsWith('ECB-'))return matchingFromId(id);const variant=variants.find(item=>item.questionId===id);if(!variant)throw Error(`Unknown electron question: ${id}`);return variant;}
