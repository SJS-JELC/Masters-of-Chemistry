import type {EnergyProfileState} from '../../contracts/editors.ts';
import type {ProfileOptions} from '../../activities/igcse/energy-enthalpy/types.ts';
import {endY} from './index.ts';
const esc=(s:string)=>s.replace(/[&<>"']/g,c=>({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c]??c));
export function profileSVG(m:EnergyProfileState,e:Partial<ProfileOptions>={}):string {
 const curve=(r:number,p:number,peak:number,dash='')=>`<path d="M100 ${r} H210 C250 ${r} 255 ${peak} 320 ${peak} S380 ${p} 410 ${p} H525" fill="none" stroke="#067b9c" stroke-width="3" ${dash}/>`;
 return `<svg xmlns="http://www.w3.org/2000/svg" style="display:block;width:100%;height:auto" viewBox="0 0 640 415" role="img" aria-label="Reaction energy profile"><style>text{font:15px sans-serif;fill:#15243b}</style><defs><marker id="ea" viewBox="0 0 10 10" refX="10" refY="5" markerWidth="6" markerHeight="6" orient="auto"><path d="M0 0L10 5L0 10Z" fill="#856000"/></marker><marker id="delta" viewBox="0 0 10 10" refX="10" refY="5" markerWidth="6" markerHeight="6" orient="auto"><path d="M0 0L10 5L0 10Z" fill="#006944"/></marker></defs><rect width="640" height="415" fill="white"/><path d="M65 32V365H605" stroke="#15243b" fill="none" stroke-width="2"/><text transform="translate(24 255) rotate(-90)">${esc(m.vertical||'Axis?')}</text><text x="205" y="404">${esc(m.horizontal||'Axis?')}</text>${e.type==='catalyst'||e.overlay?curve(220,300,85,'stroke-dasharray="9 7"'):''}${e.type==='levels'?'':curve(m.r,m.p,m.peak)}<path d="M100 ${m.r}H210 M410 ${m.p}H525" fill="none" stroke="#067b9c" stroke-width="3"/><text x="100" y="${m.r-13}">${esc(m.left||'Unlabelled')}</text><text x="410" y="${m.p-13}">${esc(m.right||'Unlabelled')}</text>${e.letters?`<text x="330" y="${m.peak-16}">B</text>`:''}${Object.entries(m.arrows).map(([name,a])=>a?`<line x1="${a.x}" x2="${a.x}" y1="${endY(m,a.tail)}" y2="${endY(m,a.head)}" stroke="${name==='ea'?'#856000':'#006944'}" stroke-width="3" marker-end="url(#${name})"/><text x="${a.x+10}" y="${(endY(m,a.tail)+endY(m,a.head))/2-5}">${e.numberArrows?(name==='ea'?'1':'2'):name==='ea'?'Eₐ':'ΔH'}</text>`:'').join('')}${e.type==='catalyst'?`<text x="385" y="35">– – Original</text><text x="385" y="62">— ${esc(m.pathLabel||'New pathway')}</text>`:''}${e.overlay?'<text x="385" y="35">Original pathway P</text><text x="385" y="62">Lower pathway Q</text>':''}</svg>`;
}
export const svgImage=(svg:string)=>`data:image/svg+xml;charset=utf-8,${encodeURIComponent(svg)}`;
