export function codec(prefix:string,fields:readonly Field[]){
 const capacity=36**6,modulus=fields.reduce((n,f)=>n*f.radix,1),seeds=Math.floor(capacity/modulus);
 return {encode(config:Readonly<Record<string,unknown>>,random:number){let code=0,place=1;for(const field of fields){const index=field.values.indexOf(config[field.key] as string|number);if(index<0)throw Error('Unsupported historical field '+field.key);code+=index*place;place*=field.radix;}return `${prefix}-${(((random>>>0)%seeds)*modulus+code).toString(36).toUpperCase().padStart(6,'0')}`;},
 decode(id:string){const match=id.trim().toUpperCase().match(new RegExp(`^${prefix}-([0-9A-Z]{6})$`));if(!match)throw Error('Invalid historical question code.');const seed=parseInt(match[1]!,36);if(seed>=capacity)throw Error('Invalid question seed.');let code=seed%modulus;const config:Record<string,string|number>={};for(const field of fields){const index=code%field.radix;code=Math.floor(code/field.radix);const value=field.values[index];if(value===undefined)throw Error('Unavailable historical field.');config[field.key]=value;}return {seed,config,id:`${prefix}-${match[1]}`};}};
}
