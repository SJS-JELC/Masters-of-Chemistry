import type {ContentBlock} from '../../contracts/question.ts';
import type {CalculationBlock} from './types.ts';
export const text=(text:string):ContentBlock=>({kind:'text',text});
export function svgImage(svg:string,alt:string):ContentBlock {
 // Original thermometer SVG is inline HTML, so its namespace was implicit. An img needs XML's explicit SVG namespace.
 const namespaced=/<svg\b[^>]*\bxmlns=/.test(svg)?svg:svg.replace(/<svg\b/,'<svg xmlns="http://www.w3.org/2000/svg"');
 return {kind:'image',src:'data:image/svg+xml;charset=utf-8,'+encodeURIComponent(namespaced.replace(/currentColor/g,'#17212b')),alt};
}
export function calculationBlocks(blocks:readonly CalculationBlock[]):ContentBlock[]{return blocks.flatMap(block=>block.type==='text'?[text(block.text)]:block.rows.map(([left,right])=>({kind:'formula' as const,text:`${left}${left?' = ':''}${typeof right==='string'?right:`(${right.fraction[0]}) ÷ (${right.fraction[1]})`}`})));}
// Original NumericMastery.mark: half of one unit in the third significant figure.
export function numericalTolerance(expected:number){return expected===0?1e-9:.500001*10**(Math.floor(Math.log10(Math.abs(expected)))-2);}
export function parseNumeric(raw:string){const text=raw.trim().replaceAll('−','-');return /^[+-]?(?:\d+(?:\.\d*)?|\.\d+)(?:[eE][+-]?\d+)?$/.test(text)?Number(text):NaN;}
