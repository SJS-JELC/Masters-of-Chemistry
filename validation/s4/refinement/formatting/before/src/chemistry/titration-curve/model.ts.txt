import type {TitrationRecord} from './types.ts';
import {answer,curve} from './core.js';
const escape=(text:string)=>text.replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;').replace(/"/g,'&quot;');
export function modelSVG(q:TitrationRecord):string {
 const a=answer(q),c=curve(q,a),x=(v:number)=>70+v/q.maxVolume*690,y=(ph:number)=>350-ph/14*320;
 const path=(points:readonly {v:number;pH:number}[])=>points.map((p,i)=>`${i?'L':'M'}${x(p.v).toFixed(2)},${y(p.pH).toFixed(2)}`).join(' ');
 return `<svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 820 435" role="img"><title>Checked ${q.id} titration curve</title><desc>Initial pH ${a.initialPH.toFixed(1)}; equivalence ${a.equivalenceVolume.toFixed(1)} cm³ at pH ${a.equivalencePH.toFixed(1)}; final pH ${a.finalPH.toFixed(1)}</desc><rect width="820" height="435" fill="white"/><g font-family="sans-serif" font-size="15" fill="#17243a">${Array.from({length:8},(_,i)=>i*2).map(ph=>`<path d="M70 ${y(ph)} H760" stroke="#dae0e8"/><text x="55" y="${y(ph)+5}" text-anchor="end">${ph}</text>`).join('')}${Array.from({length:6},(_,i)=>q.maxVolume*i/5).map(v=>`<text x="${x(v)}" y="377" text-anchor="middle">${v.toFixed(1)}</text>`).join('')}<path d="M70 30 V350 H760" fill="none" stroke="#17243a" stroke-width="2"/><path d="${path(c.before)} ${path(c.after)}" fill="none" stroke="#235e96" stroke-width="3"/><path d="M${x(a.equivalenceVolume)} ${y(a.equivalencePH)} V350" stroke="#7b609e" stroke-dasharray="4 4"/><text x="415" y="410" text-anchor="middle">Volume of ${escape(q.titrantName)} added / cm³</text><text x="15" y="180">pH</text><text x="82" y="20">${escape(q.id)} · initial ${a.initialPH.toFixed(1)} · equivalence ${a.equivalenceVolume.toFixed(1)} cm³ / pH ${a.equivalencePH.toFixed(1)} · final ${a.finalPH.toFixed(1)}</text></g></svg>`;
}
export function modelImage(q:TitrationRecord){return `data:image/svg+xml;charset=utf-8,${encodeURIComponent(modelSVG(q))}`;}
