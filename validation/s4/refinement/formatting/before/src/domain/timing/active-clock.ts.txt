import type { ActiveClock, ClockSample, IdleLimitMs, TimingCheckpoint } from '../../contracts/attempt.ts';

export const IDLE_LIMITS: readonly IdleLimitMs[] = Object.freeze([60000, 180000, 300000, 600000]);
export function validTiming(value: { readonly activeMs: number; readonly idleLimitMs: number }): boolean {
  return Number.isSafeInteger(value.activeMs) && value.activeMs >= 0 && IDLE_LIMITS.some(limit => limit === value.idleLimitMs);
}
export function browserClockSample(): ClockSample {
  return { monotonicMs: performance.now(), wallMs: Date.now(), visible: !document.hidden,
    focused: document.hasFocus(), suspended: false };
}
export interface ActiveClockOptions {
  readonly attemptId: string; readonly idleLimitMs: IdleLimitMs;
  readonly saved?: TimingCheckpoint; readonly completed?: boolean;
  readonly now?: () => ClockSample;
}
/** Retains the source deadline algorithm; frequency of callbacks never defines elapsed time. */
export function createActiveClock(options: ActiveClockOptions): ActiveClock {
  if (!options.attemptId || !IDLE_LIMITS.includes(options.idleLimitMs)) throw new Error('Invalid timing question.');
  const now = options.now ?? browserClockSample;
  let lastSample = now();
  const saved = options.saved;
  const restore = saved !== undefined && validTiming(saved) && saved.attemptId === options.attemptId
    && saved.idleLimitMs === options.idleLimitMs && typeof saved.finished === 'boolean';
  let activeMs = restore ? saved.activeMs : 0;
  let finished = options.completed === true || (restore && saved.finished);
  const known = options.completed !== true || restore;
  let running = !finished && lastSample.visible && lastSample.focused;
  let deadline = lastSample.monotonicMs + options.idleLimitMs;
  function sample(next: ClockSample): void {
    if (!Number.isFinite(next.monotonicMs) || !Number.isFinite(next.wallMs)) throw new Error('Invalid clock sample.');
    const monoGap = next.monotonicMs - lastSample.monotonicMs;
    const wallGap = next.wallMs - lastSample.wallMs;
    const suspended = next.suspended || monoGap > 5000 || wallGap > 5000 || monoGap < 0 || wallGap < 0;
    if (running && next.visible && next.focused && !suspended && !finished) {
      activeMs += Math.max(0, Math.min(next.monotonicMs, deadline) - lastSample.monotonicMs);
    }
    if (!next.visible || !next.focused || suspended || next.monotonicMs >= deadline) running = false;
    lastSample = next;
  }
  function checkpoint(): TimingCheckpoint | undefined {
    return known ? Object.freeze({ attemptId: options.attemptId, activeMs: Math.floor(activeMs), idleLimitMs: options.idleLimitMs, finished }) : undefined;
  }
  return Object.freeze({ attemptId: options.attemptId, sample,
    interact(next: ClockSample) {
      sample(next);
      if (next.visible && next.focused && !finished) { running = true; deadline = next.monotonicMs + options.idleLimitMs; }
    },
    pause(next: ClockSample) { sample(next); running = false; },
    checkpoint,
    finish(next: ClockSample) {
      sample(next); finished = true; running = false;
      const value = checkpoint();
      return value ? Object.freeze({ activeMs: value.activeMs, idleLimitMs: value.idleLimitMs }) : undefined;
    },
  });
}
