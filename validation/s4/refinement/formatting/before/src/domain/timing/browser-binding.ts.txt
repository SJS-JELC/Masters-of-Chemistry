import type { AttemptController, StudentAttempt } from '../../contracts/attempt.ts';
import { browserClockSample } from './active-clock.ts';

export interface BrowserTimingBindingOptions {
  readonly controller: AttemptController;
  /** Host queues repository writes and displays failures; this module never saves directly. */
  readonly onCheckpoint: (state: StudentAttempt) => void;
}
/** Bind once per mounted attempt, dispose on pause/navigation/unmount. No identity creation. */
export function bindBrowserAttemptTiming(options: BrowserTimingBindingOptions): () => void {
  const { controller } = options;
  let lastSave = Date.now(), signature = '';
  const persist = () => {
    const state = controller.checkpoint();
    const next = JSON.stringify(state.phase === 'answering' ? state.timing : { phase: state.phase, attemptId: state.attemptId });
    if (next !== signature) { signature = next; options.onCheckpoint(state); }
    lastSave = Date.now();
  };
  const pause = () => { controller.dispatch({ kind: 'pause' }); persist(); };
  const interaction = (event: Event) => {
    if (!event.isTrusted || (event instanceof PointerEvent && event.type === 'pointermove' && !event.buttons)) return;
    controller.interact(browserClockSample());
  };
  const hidden = () => { if (document.hidden) pause(); };
  const blur = () => { if (document.hidden || !document.hasFocus()) pause(); };
  const eventTypes = ['keydown','input','pointerdown','pointermove','touchmove','wheel','scroll'] as const;
  for (const event of eventTypes) document.addEventListener(event,interaction,{ capture: true, passive: true });
  document.addEventListener('visibilitychange',hidden);
  window.addEventListener('pagehide',pause);
  window.addEventListener('blur',blur);
  controller.sample(browserClockSample());
  const timer = window.setInterval(() => { controller.sample(browserClockSample()); if (Date.now()-lastSave >= 5000) persist(); },1000);
  return () => {
    window.clearInterval(timer);
    for (const event of eventTypes) document.removeEventListener(event,interaction,{ capture: true });
    document.removeEventListener('visibilitychange',hidden);
    window.removeEventListener('pagehide',pause); window.removeEventListener('blur',blur);
    pause();
  };
}
