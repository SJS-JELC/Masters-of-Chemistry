import {DiagramViewport} from '../../ui/DiagramViewport.tsx';
import {useRef,useState} from 'react';
import type {PointerEvent,KeyboardEvent} from 'react';
import type {EditorSurfaceProps} from '../../ui/EditorFrame.tsx';
import type {TitrationCurveState} from '../../contracts/editors.ts';
import {getRecord} from '../../activities/alevel/ph-titration-curves/provider.ts';
import {curve,pieces,indicators} from '../../chemistry/titration-curve/core.js';
import {constrain,selectPiece} from '../../chemistry/titration-curve/engine.ts';
import './titration.css';
type Anchor='initialPH'|'equivalenceVolume'|'finalPH';
const X0=70,X1=760,Y0=28,Y1=360;
export function TitrationCurveEditor({part,response,readOnly,onResponse}:EditorSurfaceProps){
 if(part.kind!=='titration-curve')throw Error('Titration editor needs a titration part.');
 const q=getRecord(part.markingPolicyId.replace(/^titration:/,''));
 const [preview,setPreview]=useState<TitrationCurveState|null>(null);
 const state=preview??(response?.kind==='titration-curve'?response:part.initial);
 const latest=useRef(state);latest.current=state;
 const drag=useRef<{field:Anchor;pointer:number}|null>(null);
 const svg=useRef<SVGSVGElement>(null);const [history,setHistory]=useState<readonly TitrationCurveState[]>([]);
 const commit=(patch:Partial<TitrationCurveState>,record=true)=>{if(readOnly)return;const old=latest.current,next=constrain(q,{...old,...patch});latest.current=next;if(record){setHistory(h=>[...h.slice(-99),old]);onResponse(next);}else setPreview(next);};
 const x=(v:number)=>X0+v/q.maxVolume*(X1-X0),y=(ph:number)=>Y1-ph/14*(Y1-Y0),plotted=curve(q,state);
 const path=(points:readonly {v:number;pH:number}[])=>points.map((p,i)=>`${i?'L':'M'}${x(p.v).toFixed(2)},${y(p.pH).toFixed(2)}`).join(' ');
 function pointer(event:PointerEvent<SVGSVGElement>){const current=drag.current,matrix=svg.current?.getScreenCTM();if(!current||current.pointer!==event.pointerId||!matrix)return;const p=new DOMPoint(event.clientX,event.clientY).matrixTransform(matrix.inverse());commit({[current.field]:current.field==='equivalenceVolume'?(p.x-X0)/(X1-X0)*q.maxVolume:(Y1-p.y)/(Y1-Y0)*14},false);}
 function key(event:KeyboardEvent<SVGCircleElement>,field:Anchor){const increase=field==='equivalenceVolume'?'ArrowRight':'ArrowUp',decrease=field==='equivalenceVolume'?'ArrowLeft':'ArrowDown';if(event.key!==increase&&event.key!==decrease)return;event.preventDefault();commit({[field]:state[field]+(event.key===increase?1:-1)*(field==='equivalenceVolume'?.5:.1)});}
 const handle=(field:Anchor,label:string,cx:number,cy:number)=><circle className="tc-handle" data-anchor={field} cx={cx} cy={cy} r={8} tabIndex={readOnly?-1:0} role="slider" aria-label={label} aria-disabled={readOnly} aria-valuenow={state[field]} aria-valuemin={field==='equivalenceVolume'?.5:0} aria-valuemax={field==='equivalenceVolume'?q.maxVolume-.5:14} aria-valuetext={field==='equivalenceVolume'?`${state[field].toFixed(1)} cm³`:`pH ${state[field].toFixed(1)}`} onKeyDown={e=>{if(!readOnly)key(e,field);}} onPointerDown={e=>{if(readOnly)return;e.preventDefault();setHistory(h=>[...h.slice(-99),state]);drag.current={field,pointer:e.pointerId};svg.current?.setPointerCapture(e.pointerId);}}/>;
 return <div className="titration-workspace"><div className="tc-sections">{(['before','after'] as const).map(side=><fieldset key={side} disabled={readOnly}><legend>{side==='before'?'Before':'After'} equivalence</legend><div className="tc-piece-grid">{pieces.filter(p=>p.side===side).map(piece=>{
 const low=Math.min(...piece.points.map(p=>p[1])),high=Math.max(...piece.points.map(p=>p[1]));
 const preview=piece.points.map((p,i)=>`${i?'L':'M'}${8+p[0]*104},${92-(p[1]-low)/(high-low||1)*84}`).join(' ');
 return <button key={piece.id} type="button" aria-pressed={state[side]===piece.id} aria-label={piece.label} onClick={()=>{setHistory(h=>[...h.slice(-99),state]);onResponse(selectPiece(q,state,side,piece.id));}}><svg viewBox="0 0 120 100" aria-hidden="true"><path d={preview}/></svg><span>{piece.label}</span></button>;
 })}</div></fieldset>)}</div><DiagramViewport label="Titration curve"><div className="tc-chart-scroll"><svg className="tc-chart" ref={svg} viewBox="0 0 810 425" aria-label={`pH curve; volume of ${q.titrantName} added`} onPointerMove={pointer} onPointerUp={()=>{if(drag.current)onResponse(latest.current);drag.current=null;setPreview(null);}} onPointerCancel={()=>{if(drag.current)onResponse(latest.current);drag.current=null;setPreview(null);}}>
 {Array.from({length:8},(_,i)=>i*2).map(ph=><g key={ph}><line x1={X0} x2={X1} y1={y(ph)} y2={y(ph)} className="tc-grid"/><text x={X0-12} y={y(ph)+5} textAnchor="end">{ph}</text></g>)}
 {Array.from({length:6},(_,i)=>q.maxVolume*i/5).map(v=><g key={v}><line x1={x(v)} x2={x(v)} y1={Y0} y2={Y1} className="tc-grid"/><text x={x(v)} y={Y1+24} textAnchor="middle">{v.toFixed(1)}</text></g>)}
 <path d={`M${X0},${Y0} V${Y1} H${X1}`} className="tc-axes"/><text x={20} y={190} textAnchor="middle" transform="rotate(-90 20 190)">pH</text><text x={415} y={415} textAnchor="middle">Volume of {q.titrantName} added / cm³</text>
 <path d={path(plotted.before)} className="tc-curve"/><path d={path(plotted.after)} className="tc-curve tc-after"/>
 <line x1={x(state.equivalenceVolume)} x2={x(state.equivalenceVolume)} y1={y(plotted.equivalencePH)} y2={Y1} className="tc-guide"/>
 {handle('initialPH','Initial pH anchor',X0,y(state.initialPH))}{handle('equivalenceVolume','Equivalence volume anchor',x(state.equivalenceVolume),y(plotted.equivalencePH))}{handle('finalPH','Final pH anchor',X1,y(state.finalPH))}
 </svg></div></DiagramViewport><p className="tc-readout">Initial pH {state.initialPH.toFixed(1)} · equivalence {state.equivalenceVolume.toFixed(1)} cm³ at pH {plotted.equivalencePH.toFixed(1)} · final pH {state.finalPH.toFixed(1)}</p>
 <fieldset disabled={readOnly} className="tc-numeric"><legend>Curve anchors — non-drag controls</legend>{([{field:'initialPH',label:'Initial pH'},{field:'equivalenceVolume',label:'Equivalence volume / cm³'},{field:'finalPH',label:'Final pH'}] as const).map(({field,label})=><label key={field}>{label}<input aria-label={label} type="number" min={field==='equivalenceVolume'?.5:0} max={field==='equivalenceVolume'?q.maxVolume-.5:14} step={field==='equivalenceVolume'?.5:.1} value={Number(state[field].toFixed(1))} onChange={e=>{if(e.target.value!==''&&Number.isFinite(e.target.valueAsNumber))commit({[field]:e.target.valueAsNumber});}}/></label>)}</fieldset>
 <fieldset disabled={readOnly}><legend>Indicator</legend><div className="tc-indicators">{indicators.map(i=><button type="button" key={i.id} aria-pressed={state.indicator===i.id} onClick={()=>commit({indicator:i.id})}><strong>{i.name}</strong><span>{i.range?`pH ${i.range[0].toFixed(1)}–${i.range[1].toFixed(1)}`:'No listed range fits'}</span>{i.range&&<small>{i.acidColour} in acid → {i.alkaliColour} in alkali</small>}</button>)}</div></fieldset>
 <p className="support-note">On smaller screens, swipe the graph horizontally to inspect the final anchor. Move pH anchors with ↑/↓ and the volume anchor with ←/→, or edit the values above. Choose the indicator whose entire transition range lies in the steep region.</p><button type="button" disabled={readOnly||!history.length} onClick={()=>{const old=history.at(-1);if(old){setHistory(h=>h.slice(0,-1));onResponse(old);}}}>Undo curve edit</button>
 </div>;
}
