import type {Namespace} from '../../contracts/identity.ts';
import type {LegacyImportPlan,LegacyRejectedRecord} from '../../contracts/integration.ts';
import type {ImportBatch} from '../../contracts/repository.ts';
import {LEGACY_RESULT_KEYS,parseLegacyImport,fingerprint} from '../legacy.ts';
import {safeJson,validateNamespace,canonical} from '../validation.ts';
export const C3_LEGACY_KEY='sjs:c3l6:2012-q2:draft:v1';
export const LEGACY_IMPORT_KEYS=[...LEGACY_RESULT_KEYS,C3_LEGACY_KEY] as const;
/** Pure, per-record planning: retain every rejected input, never clean it away. */
export async function planLegacyImport(namespace:Namespace,snapshots:Readonly<Record<string,string>>):Promise<LegacyImportPlan>{
 validateNamespace(namespace);
 const batches:ImportBatch[]=[],rejected:LegacyRejectedRecord[]=[],notes:string[]=[];
 const seen=new Map<string,string>();
 for(const [sourceKey,rawText] of Object.entries(snapshots).sort(([a],[b])=>{
  const priority=(key:string)=>LEGACY_IMPORT_KEYS.indexOf(key as typeof LEGACY_IMPORT_KEYS[number]);
  return (priority(a)<0?100:priority(a))-(priority(b)<0?100:priority(b))||a.localeCompare(b);
 })){
  const reject=(raw:unknown,reason:string,sourceId='payload')=>rejected.push({sourceKey,sourceId,raw,reason});
  let raw:unknown;
  try{if(typeof rawText!=='string'||rawText.length>10000000)throw Error('Payload exceeds 10 MB text limit.');raw=JSON.parse(rawText);safeJson(raw);}catch(e){reject(rawText,e instanceof Error?e.message:'Invalid JSON payload.');continue;}
  if(sourceKey===C3_LEGACY_KEY){
   if(namespace.course!=='alevel'){reject(raw,'C3L6 belongs to the A Level Olympiad strand.');continue;}
   const {revalidateC3Legacy}=await import('../../activities/olympiad/c3l6/legacy.ts');
   const checked=revalidateC3Legacy(raw,namespace.profileId);
   if(!checked.accepted||!checked.progress){reject(raw,checked.issues.join(' '));continue;}
   batches.push({namespace,source:{key:sourceKey,fingerprint:fingerprint(rawText)},curriculum:[],olympiad:[checked.progress]});
   notes.push('C3L6 answers and graphs were chemically revalidated; completion remains separate from curriculum progress.');
   if(checked.absentFields.length)notes.push(`C3L6 absent historical fields: ${checked.absentFields.join(', ')}. No completion was inferred from absent fields.`);
   continue;
  }
  if(!LEGACY_RESULT_KEYS.some(key=>key===sourceKey)){reject(raw,'Unsupported historical store: scheduler drafts, preferences and excluded activities are retained but cannot create independent curriculum evidence.');continue;}
  if(!Array.isArray(raw)){reject(raw,'Legacy result store must be an array.');continue;}
  const curriculum:ImportBatch['curriculum'][number][]=[];
  const sourceSeen=new Map<string,string>();
  raw.forEach((value,index)=>{
   const sourceId=typeof value==='object'&&value!==null&&'id' in value&&typeof value.id==='string'?value.id:`row-${index}`;
   const result=parseLegacyImport({namespace,sourceKey,rawText:JSON.stringify([value])});
   if(!result.ok){reject(value,result.error.message,sourceId);return;}
   const item=result.value.curriculum[0]!;
   const comparable:Record<string,unknown>={...item};delete comparable.sourceKey;
   const identity=canonical(comparable),localPrevious=sourceSeen.get(item.id),previous=seen.get(item.id);
   if(localPrevious!==undefined){if(localPrevious!==identity)reject(value,'Conflicting duplicate attempt ID; first valid source record retained.',sourceId);else notes.push(`Duplicate ${sourceId} within ${sourceKey} retains the same first evidence.`);return;}
   if(previous!==undefined&&previous!==identity){reject(value,'Conflicting duplicate attempt ID across source stores; first valid source record retained.',sourceId);return;}
   if(previous!==undefined)notes.push(`Repeated ${sourceId} across source stores retains the same first evidence; the repository deduplicates the original ID.`);
   sourceSeen.set(item.id,identity);seen.set(item.id,identity);curriculum.push(item);
  });
  // Receipt identity distinguishes reviewed accepted projections of unchanged
  // raw input. Raw archive identity remains the exact input fingerprint.
  const projection=canonical(curriculum.slice().sort((a,b)=>a.id.localeCompare(b.id)));
  batches.push({namespace,source:{key:sourceKey,fingerprint:`${fingerprint(rawText)}-accepted-${fingerprint(projection)}`},curriculum,olympiad:[]});
 }
 notes.push('Historical stores contain scores and dates, not first responses or raw marks. Missing responses and durations remain absent. Acid V1 evidence is retained under its original progression and does not become V2 mastery.');
 notes.push('Raw archives retain the exact source fingerprint. Import receipts combine that fingerprint with the accepted-record projection; equal cross-store rows keep their own stable source batches and are deduplicated by original attempt ID.');
 return {batches,rejected,notes};
}
