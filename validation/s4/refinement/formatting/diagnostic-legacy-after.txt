import { jsx as _jsx, jsxs as _jsxs } from "react/jsx-runtime";
import { useEffect, useMemo, useState } from 'react';
import { createChemistryRepository } from '../persistence/index.ts';
import { planLegacyImport, LEGACY_IMPORT_KEYS } from '../persistence/legacy/plan.ts';
import { executeLegacyImport, readableLegacyKeys, IMPORT_REPORT_KEY, } from './import.ts';
import './legacy-import.css';
export function LegacyImportView({ course, namespace, databaseName }) {
    const repository = useMemo(() => createChemistryRepository(databaseName), [databaseName]);
    const [sourceKey, setSourceKey] = useState(course === 'alevel' ? LEGACY_IMPORT_KEYS[0] : LEGACY_IMPORT_KEYS[2]);
    const [input, setInput] = useState(''), [snapshots, setSnapshots] = useState({});
    const [plan, setPlan] = useState(null), [result, setResult] = useState(null);
    const [status, setStatus] = useState(''), [busy, setBusy] = useState(false), [archives, setArchives] = useState([]);
    async function refresh() {
        const r = await repository.importArchives(namespace);
        if (r.ok)
            setArchives(r.value);
        else
            setStatus(r.error.message);
    }
    useEffect(() => {
        void refresh();
    }, [repository, namespace.course, namespace.profileId]);
    async function preview(next) {
        setBusy(true);
        setResult(null);
        setStatus('');
        try {
            setSnapshots(next);
            setPlan(await planLegacyImport(namespace, next));
            setStatus(Object.keys(next).length
                ? 'Preview ready. Import requires your next choice.'
                : 'No legacy keys were found on this origin. Paste or upload exported data from the original site.');
        }
        catch (e) {
            setStatus(String(e));
        }
        finally {
            setBusy(false);
        }
    }
    async function read() {
        try {
            const next = {};
            for (const key of readableLegacyKeys(course)) {
                const text = localStorage.getItem(key);
                if (text !== null)
                    next[key] = text;
            }
            await preview(next);
        }
        catch {
            setStatus('Legacy storage could not be read. Paste or upload an export.');
        }
    }
    async function save() {
        setBusy(true);
        try {
            const r = await executeLegacyImport(repository, namespace, snapshots);
            setResult(r);
            setStatus(r.errors.length
                ? 'Some records could not be imported. The exact sources that saved remain archived; retry or export the report.'
                : 'Import complete. Original stores were left untouched.');
            await refresh();
        }
        catch (e) {
            setStatus(String(e));
        }
        finally {
            setBusy(false);
        }
    }
    function download(value, name) {
        const blob = new Blob([JSON.stringify(value, null, 2)], { type: 'application/json' });
        const url = URL.createObjectURL(blob);
        const a = document.createElement('a');
        a.href = url;
        a.download = name;
        a.click();
        URL.revokeObjectURL(url);
    }
    const count = plan?.batches.reduce((n, b) => n + b.curriculum.length + b.olympiad.length, 0) ?? 0;
    const sourceArchives = archives.filter((a) => a.source.key !== IMPORT_REPORT_KEY), outcomeReports = archives.filter((a) => a.source.key === IMPORT_REPORT_KEY);
    return (_jsxs("section", { className: "legacy-import", "aria-labelledby": "import-heading", children: [_jsx("h2", { id: "import-heading", children: "Import original progress" }), _jsx("p", { children: "Choose a source and preview it before importing into this course and profile. Original scores, dates, optional flags and valid timing are preserved. Missing answers, marks and timing remain unknown." }), _jsx("p", { children: "Only original result stores and chemically checked C3L6 drawings can become progress. Saved schedulers and practice drafts are archived for inspection; they cannot supply missing first responses or independent timing. Rocket Recall records are retained in the report and excluded from progress." }), _jsx("button", { disabled: busy, onClick: () => void read(), children: "Read original keys on this site" }), _jsx("p", { className: "import-limits", children: "This button reads only inventoried keys on the current browser origin. It cannot access another site's storage. Original-site exports can be pasted or uploaded below. Maximum 10 MB per source." }), _jsxs("label", { children: ["Original store key", _jsx("select", { value: sourceKey, onChange: (e) => setSourceKey(e.target.value), children: readableLegacyKeys(course).map((key) => (_jsx("option", { children: key }, key))) })] }), _jsxs("label", { children: ["Source JSON", _jsx("textarea", { value: input, maxLength: 10000000, onChange: (e) => setInput(e.target.value), rows: 6 })] }), _jsxs("label", { children: ["Upload source JSON", _jsx("input", { type: "file", accept: ".json,application/json", disabled: busy, onChange: async (e) => {
                            const file = e.target.files?.[0];
                            if (!file)
                                return;
                            if (file.size > 10000000) {
                                setStatus('File exceeds 10 MB.');
                                return;
                            }
                            setInput(await file.text());
                        } })] }), _jsx("button", { disabled: busy || !input, onClick: () => void preview({ [sourceKey]: input }), children: "Preview source" }), _jsx("p", { role: "status", children: status }), plan && (_jsxs("div", { children: [_jsxs("h3", { children: ["Preview: ", count, " supported records, ", plan.rejected.length, " retained exclusions"] }), _jsx("ul", { children: plan.notes.map((note, i) => (_jsx("li", { children: note }, i))) }), plan.rejected.length > 0 && (_jsxs("details", { open: true, children: [_jsx("summary", { children: "Records excluded from progress" }), _jsx("ul", { children: plan.rejected.slice(0, 100).map((r, i) => (_jsxs("li", { children: [_jsxs("strong", { children: [r.sourceKey, " \u00B7 ", r.sourceId] }), ": ", r.reason] }, i))) }), plan.rejected.length > 100 && (_jsx("p", { children: "Showing the first 100. Export includes all exclusions and raw records." }))] })), _jsx("button", { disabled: busy || !Object.keys(snapshots).length, onClick: () => void save(), children: "Import reviewed sources" }), ' ', _jsx("button", { onClick: () => download({ namespace, snapshots, plan, result }, 'legacy-import-report.json'), children: "Export exact sources and report" })] })), result && (_jsxs("div", { children: [_jsx("h3", { children: "Import result" }), _jsxs("ul", { children: [result.receipts.map((r) => (_jsxs("li", { children: [r.source.key, ": ", r.inserted, " inserted, ", r.duplicates, " already present"] }, r.source.key))), result.errors.map((e, i) => (_jsx("li", { children: e }, i)))] })] })), _jsxs("h3", { children: ["Saved source archives (", sourceArchives.length, ")"] }), _jsx("p", { children: "These sources are stored in the new database and survive refresh. Original text is retained even when no record is imported." }), _jsx("ul", { children: sourceArchives.map((a) => (_jsxs("li", { children: [_jsx("strong", { children: a.source.key }), " \u00B7 ", a.skipped.length, " excluded records", ' ', _jsx("button", { onClick: () => download(a, 'legacy-source-archive.json'), children: "Export archive" }), a.skipped.length > 0 && (_jsxs("details", { children: [_jsx("summary", { children: "Reasons" }), _jsx("ul", { children: a.skipped.slice(0, 100).map((r, i) => (_jsxs("li", { children: [r.sourceId, ": ", r.reason] }, i))) })] }))] }, `${a.source.key}:${a.source.fingerprint}`))) }), _jsxs("h3", { children: ["Saved import outcome reports (", outcomeReports.length, ")"] }), _jsx("p", { children: "Each distinct result references the exact source archives and retains exclusions, conflicts and save errors. Repeating an identical result keeps one report." }), _jsx("ul", { children: outcomeReports.map((a) => (_jsxs("li", { children: [_jsx("ul", { children: a.notes.map((note, i) => (_jsx("li", { children: note }, i))) }), _jsx("button", { onClick: () => download(a, 'legacy-import-outcome.json'), children: "Export outcome report" }), a.skipped.length > 0 && (_jsxs("details", { children: [_jsx("summary", { children: "Outcome exclusions" }), _jsx("ul", { children: a.skipped.slice(0, 100).map((r, i) => (_jsxs("li", { children: [r.sourceId, ": ", r.reason] }, i))) })] }))] }, a.source.fingerprint))) })] }));
}
