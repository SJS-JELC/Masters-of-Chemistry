import {useCallback,useEffect,useMemo,useState} from 'react';
import type {CSSProperties} from 'react';
import type {CourseDataViewProps} from '../contracts/integration.ts';
import type {CurriculumEvidence} from '../contracts/session.ts';
import {createChemistryRepository} from '../persistence/repository.ts';
import {buildStatistics,levelOf,outcomeOf,type StatisticsModel,type StatisticsOptions} from './model.ts';
import './statistics.css';

const outcomeLabels={correct:'Correct',partial:'Partly correct',incorrect:'Incorrect'};
const date=(value:number)=>new Date(value).toLocaleDateString('en-GB',{day:'numeric',month:'short',year:'numeric'});
export function duration(value:number|null):string {
  if(value===null)return 'Not recorded';
  if(value<1000)return `${Math.round(value)} ms`;
  const seconds=Math.floor(value/1000),minutes=Math.floor(seconds/60),hours=Math.floor(minutes/60);
  return hours?`${hours}h ${minutes%60}m ${seconds%60}s`:minutes?`${minutes}m ${seconds%60}s`:`${seconds}s`;
}
function HistoryRecord({record,model}:{readonly record:CurriculumEvidence;readonly model:StatisticsModel}){
  const row=model.rows.find(item=>item.gem.id===record.gemId||item.gem.mastery.historicalAliases.includes(record.gemId));
  const assessment=record.firstAssessment;
  return <details className="statistics-history-record" data-evidence-id={record.id}>
    <summary><span className={`statistics-outcome ${outcomeOf(record)}`}>{outcomeLabels[outcomeOf(record)]}</span><strong>{row?.gem.label??record.gemId}</strong><span>Level {levelOf(record)} · {date(record.completedAt)}</span><span>{record.timing?duration(record.timing.activeMs):'Time not recorded'}</span></summary>
    <dl><dt>Question code</dt><dd><code>{record.ref?.questionId??(record.provenance==='legacy-import'?record.sourceQuestionId:undefined)??'Not recorded'}</code></dd><dt>Assessment</dt><dd>{assessment?.kind==='self-rubric'?'Student rubric review':assessment?.kind==='self-drawing'?'Student drawing review':record.selfAssessed?'Student self-review':assessment?'Marked answer':'Historical saved score'}</dd><dt>Assistance</dt><dd>{record.assisted===true?'Assisted historical assessment (saved score retained)':record.assisted===false?'Unassisted first assessment':'Not recorded in historical data'}</dd><dt>Raw marks</dt><dd>{assessment?`${assessment.marks.earned} / ${assessment.marks.available}`:'Not recorded; the saved mastery score is not a raw mark'}</dd><dt>Saved mastery score</dt><dd>{record.score}</dd><dt>Recorded active time</dt><dd>{record.timing?`${duration(record.timing.activeMs)} (${record.timing.activeMs} ms); idle allowance ${record.timing.idleLimitMs/60000} min`:'Not recorded; excluded from time totals and median'}</dd><dt>Attempt ID</dt><dd><code>{record.id}</code></dd></dl>
    {assessment&&<table><caption>First assessment mark points</caption><thead><tr><th>Part / point</th><th>Marks</th></tr></thead><tbody>{assessment.marks.points.map(point=><tr key={`${point.partId}:${point.pointId}`}><td>{point.partId} / {point.pointId}</td><td>{point.earned} / {point.available}</td></tr>)}</tbody></table>}
    {record.firstResponse?<details><summary>Saved first response</summary><p>Frozen at first submission. Later corrections do not replace this response.</p><pre>{JSON.stringify(record.firstResponse.responses,null,2)}</pre></details>:<p>First response details were not saved in this historical record.</p>}
  </details>;
}
/** Content panel: its parent provides course navigation and the shared shell. */
export function StatisticsView({course,namespace,databaseName,registry}:CourseDataViewProps){
  const repository=useMemo(()=>createChemistryRepository(databaseName),[databaseName]);
  const [history,setHistory]=useState<readonly CurriculumEvidence[]>([]),[error,setError]=useState<string|null>(null),[loading,setLoading]=useState(true);
  const [days,setDays]=useState<StatisticsOptions['days']>(30),[gemId,setGemId]=useState(''),[limit,setLimit]=useState(20),[selectedDay,setSelectedDay]=useState<number|null>(null);
  const [snapshotAt,setSnapshotAt]=useState(Date.now());
  const refresh=useCallback(async()=>{
    setLoading(true);setError(null);
    if(namespace.course!==course){setError('Statistics course and profile do not match.');setHistory([]);setLoading(false);return;}
    const result=await repository.curriculumHistory(namespace);
    if(result.ok){setHistory(result.value);setSnapshotAt(Date.now());}else{setHistory([]);setError(result.error.message);}
    setLoading(false);
  },[repository,course,namespace.course,namespace.profileId]);
  useEffect(()=>{let cancelled=false;setHistory([]);setLoading(true);setError(null);setGemId('');setSelectedDay(null);setLimit(20);
    void (async()=>{if(namespace.course!==course){if(!cancelled){setError('Statistics course and profile do not match.');setLoading(false);}return;}const result=await repository.curriculumHistory(namespace);if(cancelled)return;if(result.ok){setHistory(result.value);setSnapshotAt(Date.now());}else setError(result.error.message);setLoading(false);})();
    return()=>{cancelled=true;};
  },[repository,course,namespace.course,namespace.profileId]);
  const model=useMemo(()=>buildStatistics(namespace,registry,history,{days,...(gemId?{gemId}:{}),now:snapshotAt}),[namespace.course,namespace.profileId,registry,history,days,gemId,snapshotAt]);
  const maximum=Math.max(60000,...model.bins.map(bin=>bin.counts.activeMs));
  const selected=model.bins.find(bin=>bin.start===selectedDay);
  const records=[...model.current].reverse(),archive=[...model.historical].reverse();
  return <section className="statistics" aria-labelledby="statistics-heading">
    <header className="statistics-header"><div><p className="eyebrow">{course==='alevel'?'A Level':'IGCSE'} · {namespace.profileId}</p><h2 id="statistics-heading">Your chemistry progress</h2><p>First assessments and recorded active answering time.</p></div><button type="button" onClick={()=>void refresh()} disabled={loading}>Refresh statistics</button></header>
    {loading&&<p role="status">Loading saved progress…</p>}{error&&<p role="alert">{error} Try Refresh statistics when storage is available.</p>}
    {!loading&&!error&&<>
      <div className="statistics-filters"><label>Period <select value={days} onChange={event=>{setDays(Number(event.target.value) as StatisticsOptions['days']);setSelectedDay(null);setLimit(20);}}><option value="7">Last 7 days</option><option value="30">Last 30 days</option><option value="90">Last 90 days</option><option value="0">All time</option></select></label><label>Subtopic <select value={gemId} onChange={event=>{setGemId(event.target.value);setSelectedDay(null);setLimit(20);}}><option value="">All curriculum subtopics</option>{registry.curriculumFor(course).flatMap(activity=>activity.gems.map(gem=><option key={gem.id} value={gem.id}>{gem.label}</option>))}</select></label></div>
      <div className="statistics-metrics"><div><strong data-testid="statistics-total">{model.counts.total}</strong><span>First assessments in this period</span></div><div><strong>{model.mastered} / {model.availableLevels}</strong><span>Levels mastered · all time</span></div><div><strong data-testid="statistics-time" data-active-ms={model.counts.activeMs}>{model.counts.timed?duration(model.counts.activeMs):'Not recorded'}</strong><span>{model.counts.timed} of {model.counts.total} assessments timed</span></div><div><strong>{duration(model.counts.medianActiveMs)}</strong><span>Median time · timed assessments only</span></div></div>
      <div className="statistics-legend" aria-label="Assessment outcomes">{(['correct','partial','incorrect'] as const).map(outcome=><span key={outcome} className={`statistics-outcome ${outcome}`}>{outcomeLabels[outcome]}: {model.counts[outcome]}</span>)}</div>
      {model.counts.total===0&&<p className="statistics-empty">No current-progression assessments in this period. Complete an unassisted question or choose a longer period.</p>}
      <section aria-labelledby="statistics-time-heading"><h3 id="statistics-time-heading">Active answering time by day</h3><p>Time stops at first submission. Untimed assessments are counted above but add no estimated duration. {model.counts.total-model.counts.timed} assessment{model.counts.total-model.counts.timed===1?' has':'s have'} no recorded time.</p>
        <p className="statistics-chart-scale">Bar height: 0 to {duration(maximum)} · calendar days · scroll for the whole period</p>
        <div className="statistics-chart-scroll" tabIndex={0} aria-label="Daily recorded time chart; horizontally scrollable"><div className="statistics-chart" role="list" style={{minWidth:`${Math.max(620,model.bins.length*45)}px`}}>{model.bins.map(bin=><div role="listitem" key={bin.start} className="statistics-day"><button type="button" aria-pressed={selectedDay===bin.start} aria-label={`${date(bin.start)}: ${duration(bin.counts.activeMs)} recorded active time; ${bin.counts.total} assessments, ${bin.counts.total-bin.counts.timed} untimed. Correct ${duration(bin.counts.timeByOutcome.correct)}, partly correct ${duration(bin.counts.timeByOutcome.partial)}, incorrect ${duration(bin.counts.timeByOutcome.incorrect)}`} onClick={()=>setSelectedDay(bin.start)} data-active-ms={bin.counts.activeMs}><span className="statistics-bar" style={{height:`${bin.counts.activeMs/maximum*160}px`}}>{(['incorrect','partial','correct'] as const).map(outcome=><span key={outcome} className={outcome} style={{height:bin.counts.activeMs?`${bin.counts.timeByOutcome[outcome]/bin.counts.activeMs*100}%`:'0'}}/>)}</span><span className="statistics-day-label">{new Date(bin.start).getDate()} {new Date(bin.start).toLocaleDateString('en-GB',{month:'short'})}</span></button></div>)}</div></div>
        <p role="status" className="statistics-day-detail">{selected?`${date(selected.start)}: ${selected.counts.total} assessments; ${duration(selected.counts.activeMs)} recorded time; ${selected.counts.total-selected.counts.timed} untimed.`:'Select a day to see its totals. Each bar also has a full text description.'}</p>
      </section>
      <section aria-labelledby="statistics-mastery-heading"><h3 id="statistics-mastery-heading">Mastery by subtopic</h3><p>Mastery uses all saved current-progression evidence, regardless of the period above. Scores start from zero and weight recent questions more strongly; each level uses its original half-life. The score must be strictly above the shown threshold.</p><div className="statistics-subtopics">{model.rows.map(row=><article className="statistics-gem" key={row.gem.id}><h4>{row.gem.label}</h4><p>{row.activity.title} · {row.counts.total} assessments in this period · {row.counts.timed?duration(row.counts.activeMs):'Time not recorded'}</p><div className="statistics-levels">{row.levels.map(summary=>{
        const setting=row.gem.mastery.supportedLevels.find(band=>band.level===summary.level)!;
        return <div key={summary.level} className="statistics-level"><strong>{setting.label}</strong><div role="meter" aria-label={`${row.gem.label}, ${setting.label}, mastery`} aria-valuemin={0} aria-valuemax={100} aria-valuenow={(summary.score??0)*100} aria-valuetext={summary.score===null?'Not assessed':`${(summary.score*100).toFixed(1)} percent; ${summary.mastered?'mastered':'threshold not exceeded'}`} className={`statistics-meter${summary.mastered?' mastered':''}`} style={{'--statistics-score':`${(summary.score??0)*100}%`,'--statistics-threshold':`${row.gem.mastery.threshold*100}%`} as CSSProperties}><span/><i/></div><span>{summary.score===null?'Not assessed':`${(summary.score*100).toFixed(1)}% · ${summary.mastered?'Mastered':'In progress'}`}</span><small>{summary.count} all-time assessments · threshold &gt;{row.gem.mastery.threshold*100}% · half-life {setting.halfLife} questions</small></div>;
      })}</div></article>)}</div></section>
      <section aria-labelledby="statistics-history-heading"><h3 id="statistics-history-heading">Question history</h3><p>Most recent first assessments. Open a row for the saved raw marks and first response when available.</p>{records.slice(0,limit).map(record=><HistoryRecord key={record.id} record={record} model={model}/>)}{records.length>limit&&<button type="button" onClick={()=>setLimit(value=>value+20)}>Show 20 more assessments</button>}{model.counts.assisted>0&&<p>{model.counts.assisted} imported assessment{model.counts.assisted===1?' was':'s were'} marked assisted in the original app. Their saved scores remain unchanged.</p>}</section>
      {archive.length>0&&<details className="statistics-archive"><summary>Historical progression · {archive.length} assessments</summary><p>Retained for inspection. These records belong to an old progression or unsupported level and do not seed current mastery or the chart above.</p>{archive.slice(0,limit).map(record=><HistoryRecord key={record.id} record={record} model={model}/>)}{archive.length>limit&&<button type="button" onClick={()=>setLimit(value=>value+20)}>Show more historical assessments</button>}</details>}
      <p className="statistics-note">Extra hints and reveals before assessment create no new independent curriculum record. Corrections and worked answers after assessment keep the first score and time. Built-in level scaffolds remain valid independent practice. Olympiad challenges are outside curriculum statistics.</p>
    </>}
  </section>;
}
export default StatisticsView;
