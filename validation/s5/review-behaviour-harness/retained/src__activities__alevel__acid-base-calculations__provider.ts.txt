import type {
  ContentBlock,
  Question,
  QuestionProvider,
  QuestionRef,
  QuestionSelection,
  Level,
} from '../../../contracts/index.ts';
import { acidEngine, type AcidQuestion, type AcidResponse } from './engine.js';
import { acidSources } from './sources.ts';

export const acidActivityId = 'alevel/acid-base-calculations' as const;
/** Content-only notation conversion; no HTML/DOM or inferred formula parsing. */
export function acidNotation(text: string): string {
  return text
    .replace(/pK_a|\bpKa\b/g, 'pKₐ')
    .replace(/K_a|\bKa\b/g, 'Kₐ')
    .replace(/K_w|\bKw\b/g, 'K_w')
    .replace(/M_r/g, 'Mᵣ')
    .replace(/\^\(([−+-]?\d+)\)/g, (_match: string, power: string) => superscript(power))
    .replace(/\^([−+-]?\d+)/g, (_match: string, power: string) => superscript(power));
}
function superscript(text: string): string {
  const map: Record<string, string> = {
    '-': '⁻',
    '−': '⁻',
    '+': '⁺',
    '0': '⁰',
    '1': '¹',
    '2': '²',
    '3': '³',
    '4': '⁴',
    '5': '⁵',
    '6': '⁶',
    '7': '⁷',
    '8': '⁸',
    '9': '⁹',
  };
  return [...text].map((c) => map[c] ?? c).join('');
}
function text(value: string): ContentBlock {
  return { kind: 'text', text: acidNotation(value) };
}
function roundingInstruction(response: AcidResponse): string {
  const precision: Record<AcidResponse['format'], string> = {
    dp2: 'to 2 decimal places',
    dp3: 'to 3 decimal places',
    sf3: 'to 3 significant figures',
    sf4: 'to 4 significant figures',
    integer: 'as a whole number',
  };
  const quantity = response.unit ? `${response.symbol} (${response.unit})` : response.symbol;
  return `Give ${quantity} ${precision[response.format]}.`;
}
export function acidRef(item: AcidQuestion): QuestionRef {
  return {
    activityId: acidActivityId,
    questionId: item.reviewId,
    seed: item.seed,
    level: item.level,
  };
}
function seed(value: number): void {
  if (!Number.isSafeInteger(value) || value < 0 || value > 0xffffffff)
    throw Error('Acid seed must be an unsigned 32-bit integer.');
}
export function acidSourceQuestion(ref: QuestionRef): AcidQuestion {
  if (ref.activityId !== acidActivityId) throw Error('Acid activity identity mismatch.');
  seed(ref.seed);
  const item = acidEngine.generateFromReview(ref.questionId);
  if (item.reviewId !== ref.questionId || item.seed !== ref.seed || item.level !== ref.level)
    throw Error('Acid question code, seed and level disagree.');
  return item;
}
export function acidQuestion(item: AcidQuestion): Question {
  const rows: ContentBlock[] = item.rows.length
    ? [
        {
          kind: 'table',
          headers: ['Quantity', 'Value'],
          rows: item.rows.map((row) => [acidNotation(row.label), acidNotation(row.value)]),
        },
      ]
    : [];
  return {
    ref: acidRef(item),
    title: acidNotation(item.templateLabel),
    context: [
      text(item.intro),
      ...rows,
      text(
        'Use the rounding instructions shown with each response. You can enter standard form as 1.23e-4.',
      ),
      ...(item.legacy
        ? [
            text(
              'Historical question for review. Its former progression does not count towards current mastery.',
            ),
          ]
        : []),
    ],
    layout: item.responses.length > 1 ? 'multipart' : 'compact',
    submission: 'all-required-parts',
    parts: item.responses.map((response) => ({
      id: response.key,
      kind: 'numeric',
      inputMode: 'text',
      prompt: [
        text(response.prompt),
        text(roundingInstruction(response)),
        { kind: 'formula', text: `${acidNotation(response.symbol)} =` },
      ],
      marks: 1,
      required: true,
      dependsOn: [],
      unit: response.unit,
      acceptance: { kind: 'absolute', expected: response.expected, tolerance: response.tolerance },
    })),
    scaffolds:
      item.level === 1
        ? [
            {
              id: 'structured-steps',
              level: 1,
              purpose: 'Guided calculation stages',
              content: [
                text(
                  'Work through the numerical stages in order. Each stage asks for a quantity needed in the calculation.',
                ),
              ],
            },
          ]
        : [],
    hints: [],
    workedAnswer: item.working
      .flatMap((line) => line.split(';'))
      .map((line) => ({ kind: 'formula' as const, text: acidNotation(line.trim()) })),
    sources: acidSources,
  };
}
export function validateAcidSelection(selection: QuestionSelection): void {
  if (
    selection.activityId !== acidActivityId ||
    !selection.gemId ||
    !acidEngine.scopeFor(selection.gemId) ||
    !acidEngine.templatesFor(selection.gemId, selection.level).length
  )
    throw Error('Unsupported acid gem or level.');
  seed(selection.seed);
  if (!Array.isArray(selection.previousQuestionIds))
    throw Error('Previous acid identities must be an array.');
  for (const id of selection.previousQuestionIds) {
    const item = acidEngine.generateFromReview(id);
    if (item.legacy) throw Error('Historical progression cannot select current mastery questions.');
  }
}
export const acidProvider: QuestionProvider = {
  coverage: [
    {
      kind: 'generated',
      families: acidEngine.templates.map((template) => ({
        templateId: template.id,
        levels: ([1, 2, 3] as const).filter((level) =>
          acidEngine.scopes.some((scope) => scope.levels[level].includes(template.id)),
        ),
      })),
    },
  ],
  select(selection) {
    validateAcidSelection(selection);
    const gem = selection.gemId!;
    const used = selection.previousQuestionIds
      .map((id) => acidEngine.decodeV2(id))
      .filter((item) => item.level === selection.level)
      .map((item) => item.templateId);
    // Original chooseTemplate cycles by template family; seed determines the draw explicitly.
    let value = selection.seed;
    value = Math.imul(value ^ (value >>> 16), 0x45d9f3b) >>> 0;
    value = Math.imul(value ^ (value >>> 16), 0x45d9f3b) >>> 0;
    const chosen = acidEngine.chooseTemplate(
      gem,
      selection.level,
      used,
      ((value ^ (value >>> 16)) >>> 0) / 4294967296,
    );
    return acidRef(acidEngine.generate(chosen.templateId, selection.level, selection.seed));
  },
  restore(ref) {
    return acidQuestion(acidSourceQuestion(ref));
  },
  resolveLink(code) {
    try {
      return acidRef(acidEngine.generateFromReview(code));
    } catch {
      return null;
    }
  },
};
export function acidTemplateFor(ref: QuestionRef): string {
  return acidSourceQuestion(ref).templateId;
}
export function acidAvailableLevels(templateId: string): readonly Level[] {
  return ([1, 2, 3] as const).filter((level) =>
    acidEngine.scopes.some((scope) => scope.levels[level].includes(templateId)),
  );
}
