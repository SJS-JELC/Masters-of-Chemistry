import type { EditorKind, EditorState, EditorStateFor } from './editors';
import type {
  ActivityId,
  GemId,
  Level,
  PartId,
  QuestionId,
  QuestionRef,
  Seed,
  SourceReference,
} from './identity';

export type ContentBlock =
  | Readonly<{ kind: 'text'; text: string }>
  | Readonly<{ kind: 'formula'; text: string }>
  | Readonly<{ kind: 'image'; src: string; alt: string }>
  | Readonly<{ kind: 'table'; headers: readonly string[]; rows: readonly (readonly string[])[] }>;
export interface ChoiceOption {
  readonly id: string;
  readonly content: readonly ContentBlock[];
}
export type TextRange = Readonly<{ start: number; end: number; text: string }>;
export type Response =
  | Readonly<{ kind: 'choice'; selected: readonly string[] }>
  | Readonly<{ kind: 'text'; value: string }>
  | Readonly<{ kind: 'numeric'; raw: string; unit: string }>
  | Readonly<{ kind: 'explanation'; sections: readonly Readonly<{ id: string; text: string }>[] }>
  | Readonly<{ kind: 'correction'; selections: readonly TextRange[]; replacement: string }>
  | Readonly<{ kind: 'diagram-selection'; selectedObjectIds: readonly string[] }>
  | Readonly<{ kind: 'drawing-self-check'; matches: boolean | null }>
  | EditorState;
export type ResponseKind = Response['kind'];
export type ResponseFor<K extends ResponseKind> = Extract<Response, { readonly kind: K }>;
export type Responses = Readonly<Partial<Record<PartId, Response>>>;
export type NumericAcceptance =
  | Readonly<{ kind: 'absolute'; expected: number; tolerance: number }>
  | Readonly<{
      kind: 'relative';
      expected: number;
      relativeTolerance: number;
      absoluteFloor: number;
    }>
  | Readonly<{
      kind: 'rounding';
      expected: number;
      digits: number;
      mode: 'decimal-places' | 'significant-figures';
    }>;
interface PartBase {
  readonly id: PartId;
  readonly prompt: readonly ContentBlock[];
  readonly marks: number;
  readonly required: boolean;
  readonly dependsOn: readonly Readonly<{ partId: PartId; condition: 'response-reference' }>[];
}
export interface RubricPoint {
  readonly id: string;
  readonly text: string;
  readonly reject?: string;
  readonly marks: number;
}
export interface NumericWorkingField {
  readonly id: string;
  readonly expected: number;
  readonly tolerance: number;
  readonly finalResponse?: true;
}
export type NumericWorkingToken = string | NumericWorkingField;
export interface NumericWorkingFramework {
  readonly supportId: string;
  readonly rows: readonly Readonly<{
    left: string;
    right:
      | Readonly<{ kind: 'tokens'; tokens: readonly NumericWorkingToken[] }>
      | Readonly<{
          kind: 'fraction';
          numerator: readonly NumericWorkingToken[];
          denominator: readonly NumericWorkingToken[];
        }>;
  }>[];
}
export type StandardPart = PartBase &
  (
    | Readonly<{
        kind: 'choice';
        presentation: 'single' | 'multiple' | 'dropdown';
        options: readonly ChoiceOption[];
        acceptedOptionSets: readonly (readonly string[])[];
      }>
    | Readonly<{
        kind: 'text';
        accepted: readonly string[];
        normalization: 'exact' | 'chemical-text';
        presentation?: 'single-line' | 'multiline';
      }>
    | Readonly<{
        kind: 'numeric';
        unit: string;
        acceptance: NumericAcceptance;
        inputMode?: 'decimal' | 'text';
        workingFramework?: NumericWorkingFramework;
      }>
    | Readonly<{
        kind: 'explanation';
        sections: readonly Readonly<{
          id: string;
          label: string;
          help?: string;
          presentation?: 'single-line' | 'multiline';
          group?: string;
          minLength?: number;
        }>[];
        minLength?: number;
        rubric: readonly RubricPoint[];
        assessment: 'self-rubric';
      }>
    | Readonly<{
        kind: 'correction';
        sourceText: string;
        markingPolicyId: string;
        segments?: readonly (TextRange & Readonly<{ id: string }>)[];
        segmentSelection?: 'single' | 'multiple';
      }>
    | Readonly<{
        kind: 'diagram-selection';
        diagram: readonly ContentBlock[];
        objectIds: readonly string[];
        acceptedObjectSets: readonly (readonly string[])[];
      }>
    | Readonly<{
        kind: 'drawing-self-check';
        model: readonly ContentBlock[];
        criteria: readonly string[];
      }>
  );
export type EditorPart = {
  [K in EditorKind]: PartBase &
    Readonly<{ kind: K; initial: EditorStateFor<K>; markingPolicyId: string }>;
}[EditorKind];
export type QuestionPart = StandardPart | EditorPart;
export interface Scaffold {
  readonly id: string;
  readonly level: Level;
  readonly content: readonly ContentBlock[];
  readonly purpose: string; // built-in pedagogical support; does not count as a requested hint
}
export interface Hint {
  readonly id: string;
  readonly content: readonly ContentBlock[];
}
export interface Question {
  readonly ref: QuestionRef;
  readonly title: string;
  readonly context: readonly ContentBlock[];
  readonly layout: 'compact' | 'multipart' | 'workspace';
  readonly submission: 'all-required-parts';
  readonly parts: readonly QuestionPart[];
  readonly scaffolds: readonly Scaffold[];
  readonly hints: readonly Hint[];
  readonly workedAnswer: readonly ContentBlock[];
  readonly sources: readonly SourceReference[];
}
/** Full fixed banks and every generator family must be exposed, not a sampled pilot. */
export type ProviderCoverage =
  | Readonly<{ kind: 'fixed'; questionIds: readonly QuestionId[] }>
  | Readonly<{
      kind: 'generated';
      families: readonly Readonly<{ templateId: string; levels: readonly Level[] }>[];
    }>;
export interface QuestionSelection {
  readonly activityId: ActivityId;
  readonly gemId?: GemId;
  readonly level: Level;
  readonly seed: Seed;
  readonly previousQuestionIds: readonly QuestionId[];
}
export interface QuestionProvider {
  readonly coverage: readonly ProviderCoverage[];
  readonly select: (selection: QuestionSelection) => QuestionRef;
  readonly restore: (ref: QuestionRef) => Question;
  /** Resolve supported historical review codes without deriving IDs from bank order. */
  readonly resolveLink: (code: string) => QuestionRef | null;
}
/** Source energy text states: correct/incorrect/empty/unknown. Unknown is not wrong. */
export type TextClassification = 'accepted' | 'rejected' | 'empty' | 'unrecognized';
export interface ValidAlternativeEligibility {
  readonly kind: 'valid-alternative';
  readonly modelAnswer: string;
  readonly rubric: readonly ContentBlock[];
  /** Activity policy checks nonempty free text and correction prerequisites. */
  readonly eligible: boolean;
}
export interface MarkPointResult {
  readonly partId: PartId;
  readonly pointId: string;
  readonly earned: number;
  readonly available: number;
  readonly message: string;
  readonly textClassification?: TextClassification;
  readonly learningReview?: ValidAlternativeEligibility;
}
export interface RawMarks {
  readonly earned: number;
  readonly available: number;
  readonly points: readonly MarkPointResult[];
}
export interface MarkingIssue {
  readonly partId: PartId;
  readonly message: string;
  readonly textClassification?: 'empty' | 'unrecognized';
  readonly rubricSupport?: readonly ContentBlock[];
}
export type MarkingOutcome =
  | Readonly<{ accepted: false; issues: readonly MarkingIssue[] }>
  | Readonly<{ accepted: true; marks: RawMarks }>;
/** No storage/timer/DOM dependencies; invalid/incomplete input is not an assessment. */
export interface MarkingPolicy {
  readonly id: string;
  readonly mark: (question: Question, responses: Responses) => MarkingOutcome;
  readonly masteryScore: (marks: RawMarks) => 0 | 0.5 | 1;
}
