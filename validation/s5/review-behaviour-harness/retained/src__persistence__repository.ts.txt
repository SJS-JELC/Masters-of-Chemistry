  canonical,
  record,
  requireData,

  validateAttempt,
  validateEvidence,

  validateOlympiad,
  validateSession,
  validateWrite,
} from './validation.ts';
interface Row<T> {
  course: string;
  profileId: string;
  id: string;
  value: T;
}
interface ReceiptRow {
  course: string;
  profileId: string;
  key: string;
  fingerprint: string;
  value: ImportReceipt;
  payload: string;
}
interface ArchiveRow {
  course: string;
  profileId: string;
  key: string;
  fingerprint: string;
  value: LegacyImportArchive;
}
type Identity = [string, string, string];
export const DATABASE_SCHEMA = {
  attempts: '[course+profileId+id], [course+profileId]',
  evidence: '[course+profileId+id], [course+profileId]',
  sessions: '[course+profileId+id], [course+profileId]',
  olympiad: '[course+profileId+id], [course+profileId]',
  receipts: '[course+profileId+key+fingerprint]',
};
export const DATABASE_SCHEMA_V2 = {
  ...DATABASE_SCHEMA,
  archives: '[course+profileId+key+fingerprint], [course+profileId]',
};
function validateArchive(value: unknown): asserts value is LegacyImportArchive {
  const archive = record(value, 'archive'),
    source = record(archive.source, 'archive source');
  requireData(
    Object.keys(archive).every((key) =>
      ['namespace', 'source', 'rawText', 'notes', 'skipped'].includes(key),
    ) && Object.keys(source).every((key) => ['key', 'fingerprint'].includes(key)),
    'Unexpected archive field',
  );
  validateNamespace(archive.namespace);
  requireData(
    typeof source.key === 'string' && source.key.length > 0 && source.key.length <= 500,
    'Invalid archive source key',
  );
  requireData(
    typeof source.fingerprint === 'string' &&
      source.fingerprint.length > 0 &&
      source.fingerprint.length <= 500,
    'Invalid archive fingerprint',
  );
  // Exact rejected source text can itself be malformed JSON and must remain unchanged.
  requireData(
    typeof archive.rawText === 'string' && archive.rawText.length <= 10000000,
    'Archive source exceeds bounds',
  );
  requireData(
    Array.isArray(archive.notes) &&
      archive.notes.length <= 10000 &&
      archive.notes.every((note) => typeof note === 'string' && note.length <= 10000),
    'Invalid archive notes',
  );
  requireData(
    Array.isArray(archive.skipped) && archive.skipped.length <= 100000,
    'Invalid archive skipped records',
  );
  for (const value of archive.skipped) {
    const item = record(value, 'archive skipped record');
    requireData(
      Object.keys(item).every((key) => ['sourceId', 'reason'].includes(key)) &&
        typeof item.sourceId === 'string' &&
        item.sourceId.length <= 500 &&
        typeof item.reason === 'string' &&
        item.reason.length <= 10000,
      'Invalid archive skipped record',
    );
  }
}
function identity(namespace: Namespace, id: string): Identity {
  return [namespace.course, namespace.profileId, id];
}
function row<T>(namespace: Namespace, id: string, value: T): Row<T> {
  return { course: namespace.course, profileId: namespace.profileId, id, value };
}
function failure(error: unknown): StorageFailure {
  if (error instanceof EvidenceConflict)
    return { code: 'conflict', message: error.message, retryable: false };
  if (error instanceof InvalidData)
    return { code: 'invalid-data', message: error.message, retryable: false };
  const name = error && typeof error === 'object' && 'name' in error ? String(error.name) : '';
  if (name === 'QuotaExceededError' || name === 'QuotaExceeded')
    return {
      code: 'quota',
      message: 'Storage is full. Your current work remains in memory; free space and retry.',
      retryable: true,
    };
  return {
    code: 'unavailable',
    message:
      'Browser storage is unavailable. Your current work remains in memory; retry saving when storage is available.',
    retryable: true,
  };
}
async function result<T>(operation: () => Promise<T>): Promise<RepositoryResult<T>> {
  try {
    return { ok: true, value: await operation() };
  } catch (error) {
    return { ok: false, error: failure(error) };
  }
}
function conflict(condition: unknown, message: string): void {
  if (!condition) throw new EvidenceConflict(message);
}
/** A validation-only fault point runs inside the actual IndexedDB transaction. */
export type TransactionFault = (
  point:
    | 'after-attempt'
    | 'after-evidence'
    | 'after-import-row'
    | 'before-receipt'
    | 'session'
    | 'olympiad',
) => void;
/** Production factory. Database names can never select the legacy stores. */
export function createChemistryRepository(databaseName: string): ChemistryRepository {
  return createRepository(databaseName);
}
/** Imported by retained browser fixtures only; not re-exported from the production entry. */
export function createRepository(
  databaseName: string,
  fault?: TransactionFault,
): ChemistryRepository {
  const database = new Dexie(databaseName);
  database.version(1).stores(DATABASE_SCHEMA);
  database.version(2).stores(DATABASE_SCHEMA_V2);
  const attempts = database.table<Row<StudentAttempt>, Identity>('attempts');
  const evidence = database.table<Row<CurriculumEvidence>, Identity>('evidence');
  const sessions = database.table<Row<CurriculumSession>, Identity>('sessions');
  const olympiad = database.table<Row<C3L6Progress>, Identity>('olympiad');
  const receipts = database.table<ReceiptRow, [string, string, string, string]>('receipts');
  const archives = database.table<ArchiveRow, [string, string, string, string]>('archives');
  function nameGuard(): void {
    requireData(
      /^masters-of-chemistry-[a-zA-Z0-9_-]{1,150}$/.test(databaseName),
      'Database must use the new masters-of-chemistry- namespace',
    );
  }
  async function load<T>(
    table: Table<Row<T>, Identity>,
    namespace: Namespace,
    id: string,
    validator: (v: unknown) => void,
  ): Promise<T | null> {
    nameGuard();
    validateNamespace(namespace);
    requireData(typeof id === 'string' && id.length > 0 && id.length <= 500, 'Invalid record ID');
    const saved = await table.get(identity(namespace, id));
    if (!saved) return null;
    validator(saved.value);
    return saved.value;
  }
  async function putEvidence(value: CurriculumEvidence): Promise<'inserted' | 'already-present'> {
    const namespace = { course: value.course, profileId: value.profileId };
    const previous = await evidence.get(identity(namespace, value.id));
    if (previous) {
      const left = { ...previous.value },
        right = { ...value };
      if (left.provenance === 'legacy-import' && right.provenance === 'legacy-import') {
        delete left.sourceKey;
        delete right.sourceKey;
      }
      conflict(same(left, right), 'First assessment differs from the saved immutable evidence');
      return 'already-present';
    }
    await evidence.add(row(namespace, value.id, value));
    return 'inserted';
  }
  async function saveCurriculum(write: CurriculumWrite): Promise<SaveReceipt> {
    nameGuard();
    validateWrite(write);
    // Clone before the first await: callers keep the editable draft on failure and
    // later mutations cannot alter the validated transaction input.
    const copied = structuredClone(write),
      a = copied.attempt;
    return database.transaction('rw', attempts, evidence, sessions, async () => {
      const previous = await attempts.get(identity(a.namespace, a.attemptId));
      if (previous) {
        const old = previous.value;
        conflict(
          same(a.assistance.slice(0, old.assistance.length), old.assistance),
          'Assistance history cannot be erased',
        );
        conflict(
          same(old.ref, a.ref) && same(old.target, a.target),
          'Attempt identity points to a different question',
        );
        if (old.phase !== 'answering') {
          conflict(
            a.phase !== 'answering' && same(old.firstResponse, a.firstResponse),
            'Frozen first response cannot change',
          );
          if (old.phase === 'rubric-review' || old.phase === 'drawing-review') {
            conflict(
              a.phase === old.phase ||
                (a.phase === 'assessed' &&
                  a.firstAssessment.kind ===
                    (old.phase === 'rubric-review' ? 'self-rubric' : 'self-drawing')),
              'Pending self-review cannot change assessment method',
            );
            if (a.phase === old.phase)
              conflict(
                same(old.automaticMarks, a.automaticMarks),
                'First automatic marks cannot change during self-review',
              );
            if (a.phase === 'assessed' && a.firstAssessment.kind !== 'revealed') {
              for (const point of old.automaticMarks?.points ?? []) {
                conflict(
                  a.firstAssessment.marks.points.some((saved) => same(saved, point)),
                  'Aggregate assessment changed first automatic marks',
                );
              }
            }
          }
          if (old.phase === 'assessed')
            conflict(
              a.phase === 'assessed' && same(old.firstAssessment, a.firstAssessment),
              'First assessment cannot change',
            );
        }
      }
      const previousEvidence = await evidence.get(identity(a.namespace, a.attemptId));
      if (previousEvidence) {
        conflict(
          a.phase === 'assessed' &&
            same(previousEvidence.value.firstResponse, a.firstResponse) &&
            same(previousEvidence.value.firstAssessment, a.firstAssessment),
          'Attempt conflicts with saved first evidence',
        );
      }
      await attempts.put(row(a.namespace, a.attemptId, a));
      fault?.('after-attempt');
      const status = copied.evidence === undefined ? 'none' : await putEvidence(copied.evidence);
      fault?.('after-evidence');
      if (copied.session !== undefined) {
        await sessions.put(row(a.namespace, copied.session.id, copied.session));
        fault?.('session');
      }
      return { attemptId: a.attemptId, evidence: status };
    });
  }
  async function importBatch(batch: ImportBatch): Promise<ImportReceipt> {
    nameGuard();
    validateImport(batch);
    const copied = structuredClone(batch);
    return database.transaction('rw', evidence, olympiad, receipts, async () => {
      const key: [string, string, string, string] = [
        copied.namespace.course,
        copied.namespace.profileId,
        copied.source.key,
        copied.source.fingerprint,
      ];
      const previous = await receipts.get(key);
      if (previous) {
        conflict(
          previous.payload === canonical(copied),
          'Source fingerprint collision or changed import payload',
        );
        return {
          ...previous.value,
          inserted: 0,
          duplicates: copied.curriculum.length + copied.olympiad.length,
        };
      }
      let inserted = 0,
        duplicates = 0;
      for (const item of copied.curriculum) {
        const status = await putEvidence(item);
        if (status === 'inserted') inserted++;
        else duplicates++;
        fault?.('after-import-row');
      }
      for (const item of copied.olympiad) {
        const old = await olympiad.get(identity(copied.namespace, item.activityId));
        if (old) {
          conflict(same(old.value, item), 'Imported challenge differs from saved challenge');
          duplicates++;
        } else {
          await olympiad.add(row(copied.namespace, item.activityId, item));
          inserted++;
        }
        fault?.('after-import-row');
      }
      const receipt: ImportReceipt = { source: copied.source, inserted, duplicates, skipped: [] };
      fault?.('before-receipt');
      await receipts.add({
        course: copied.namespace.course,
        profileId: copied.namespace.profileId,
        key: copied.source.key,
        fingerprint: copied.source.fingerprint,
        value: receipt,
        payload: canonical(copied),
      });
      return receipt;
    });
  }
  return {
    loadAttempt: (namespace, id) => result(() => load(attempts, namespace, id, validateAttempt)),
    loadSession: (namespace, id) => result(() => load(sessions, namespace, id, validateSession)),
    curriculumHistory: (namespace) =>
      result(async () => {
        nameGuard();
        validateNamespace(namespace);
        const values = await evidence
          .where('[course+profileId]')
          .equals([namespace.course, namespace.profileId])
          .toArray();
        for (const r of values) validateEvidence(r.value);
        return values
          .map((r) => r.value)
          .sort((a, b) => a.completedAt - b.completedAt || a.id.localeCompare(b.id));
      }),
    saveCurriculum: (write) => result(() => saveCurriculum(write)),
    saveSession: (session) =>
      result(async () => {
        nameGuard();
        validateSession(session);
        const copied = structuredClone(session);
        await database.transaction('rw', sessions, async () => {
          await sessions.put(row(copied.namespace, copied.id, copied));
          fault?.('session');
        });
        return copied.id;
      }),
    loadOlympiad: (profileId) =>
      result(() =>
        load(
          olympiad,
          { course: 'alevel', profileId },
          'alevel/c3l6-organic-reactions',
          validateOlympiad,
        ),
      ),
    saveOlympiad: (progress) =>
      result(async () => {
        nameGuard();
        validateOlympiad(progress);
        const copied = structuredClone(progress);
        await database.transaction('rw', olympiad, async () => {
          await olympiad.put(
            row({ course: 'alevel', profileId: copied.profileId }, copied.activityId, copied),
          );
          fault?.('olympiad');
        });
        return 'saved' as const;
      }),
    importBatch: (batch) => result(() => importBatch(batch)),
    saveImportArchive: (archive) =>
      result(async () => {
        nameGuard();
        validateArchive(archive);
        const copied = structuredClone(archive);
        return database.transaction('rw', archives, async () => {
          const key: [string, string, string, string] = [
            copied.namespace.course,
            copied.namespace.profileId,
            copied.source.key,
            copied.source.fingerprint,
          ];
          const old = await archives.get(key);
          if (old) {
            conflict(
              same(old.value, copied),
              'Archived source or decisions differ for the same fingerprint',
            );
            return 'already-present' as const;
          }
          await archives.add({
            course: copied.namespace.course,
            profileId: copied.namespace.profileId,
            key: copied.source.key,
            fingerprint: copied.source.fingerprint,
            value: copied,
          });
          return 'saved' as const;
        });
      }),
    importArchives: (namespace) =>
      result(async () => {
        nameGuard();
        validateNamespace(namespace);
        const values = await archives
          .where('[course+profileId]')
          .equals([namespace.course, namespace.profileId])
          .toArray();
        for (const saved of values) validateArchive(saved.value);
        return values
          .map((saved) => saved.value)
          .sort(
            (a, b) =>
              a.source.key.localeCompare(b.source.key) ||
              a.source.fingerprint.localeCompare(b.source.fingerprint),
          );
      }),
  };
}
